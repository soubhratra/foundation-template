class RequireScope:
    def __init__(self, scope: str): self.scope = scope
    def has_permission(self, request, controller) -> bool:
        if not hasattr(request, "api_key") or self.scope not in request.api_key.scopes:
            raise PermissionDenied  # maps to HTTP 403 in Django Ninja
        return True
