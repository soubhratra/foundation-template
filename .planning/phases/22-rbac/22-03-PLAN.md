    ```python
    def list_users(self, page: int = 1, page_size: int = 20):
        """Return a page of all users with their groups prefetched.

        prefetch_related("groups") is required to avoid N+1 queries —
        the global _nplusone_guard fixture in tests/conftest.py will catch
        any violation during tests.
        """
        from django.contrib.auth import get_user_model
        User = get_user_model()
        offset = (page - 1) * page_size
        return User.objects.prefetch_related("groups").order_by("id")[offset : offset + page_size]

    def get_user_roles(self, user_id: int) -> list[str]:
        from django.contrib.auth import get_user_model
        User = get_user_model()
        user = get_object_or_404(User, id=user_id)
        return [g.name for g in user.groups.all()]

    def assign_role(self, user_id: int, role: str) -> object:
        """Assign a role (Django Group) to a user. Idempotent."""
        from django.contrib.auth import get_user_model
        from django.contrib.auth.models import Group
        User = get_user_model()
        user = get_object_or_404(User, id=user_id)
        group = get_object_or_404(Group, name=role)
        user.groups.add(group)
        return user

    def remove_role(self, user_id: int, role: str) -> object:
        """Remove a role (Django Group) from a user. Idempotent — no error if not assigned."""
        from django.contrib.auth import get_user_model
        from django.contrib.auth.models import Group
        User = get_user_model()
        user = get_object_or_404(User, id=user_id)
        try:
            group = Group.objects.get(name=role)
            user.groups.remove(group)
        except Group.DoesNotExist:
            pass  # idempotent — removing a non-existent role is a no-op
        return user
    ```

    Note: `get_object_or_404` is already imported in services.py (check first; if not, import from `django.shortcuts`). Use consistent import style with the existing file.

    Full AdminController:
    ```python
    from .permissions import KNOWN_ROLES, RequireRole
    from .schemas import AdminUserOut, RoleAssignIn, UserRoleOut

    @api_controller(
        "/admin/users",
        tags=["Admin"],
        permissions=[IsAuthenticated, RequireRole("admin")],
        auth=[SessionAuth(), jwt_token_auth],
    )
    class AdminController(ControllerBase):
        def __init__(self):
            self.service = UserService()

        @route.get("", response=list[AdminUserOut], url_name="admin-user-list")
        def list_users(self, request, page: int = 1, page_size: int = 20):
            """List all users with their roles. Admin only."""
            users = self.service.list_users(page=page, page_size=page_size)
            return list(users)

        @route.get("/{user_id}/roles", response=UserRoleOut, url_name="admin-user-roles")
        def get_user_roles(self, request, user_id: int):
            """Get the current roles for a specific user. Admin only."""
            roles = self.service.get_user_roles(user_id=user_id)
            return UserRoleOut(user_id=user_id, roles=roles)

        @route.post("/{user_id}/roles", response={200: UserRoleOut, 400: dict}, url_name="admin-assign-role")
        def assign_role(self, request, user_id: int, payload: RoleAssignIn):
            """Assign a role to a user. Idempotent. Admin only."""
            if user_id == request.user.id:
                return 400, {"detail": "Cannot modify your own roles."}
            if payload.role not in KNOWN_ROLES:
                return 400, {"detail": f"Unknown role '{payload.role}'. Valid roles: {sorted(KNOWN_ROLES)}"}
            user = self.service.assign_role(user_id=user_id, role=payload.role)
            return 200, UserRoleOut(user_id=user.id, roles=[g.name for g in user.groups.prefetch_related().all()])

        @route.delete("/{user_id}/roles/{role}", response={200: UserRoleOut, 400: dict}, url_name="admin-remove-role")
        def remove_role(self, request, user_id: int, role: str):
            """Remove a role from a user. Idempotent. Admin only."""
            if user_id == request.user.id:
                return 400, {"detail": "Cannot modify your own roles."}
            if role not in KNOWN_ROLES:
                return 400, {"detail": f"Unknown role '{role}'. Valid roles: {sorted(KNOWN_ROLES)}"}
            user = self.service.remove_role(user_id=user_id, role=role)
            return 200, UserRoleOut(user_id=user.id, roles=[g.name for g in user.groups.all()])
    ```
