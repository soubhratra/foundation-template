    **apps/web/src/actions/billing-actions.ts — create server actions file:**

    ```typescript
    "use server";

    import { revalidatePath } from "next/cache";
    import { getServerApi } from "@/lib/api/server";
    import type { ActionResponse } from "@/actions/profile-actions";

    function handleActionError(error: unknown, fallback: string): ActionResponse {
      const message = error instanceof Error ? error.message : fallback;
      return { status: "error", message };
    }

    export async function changePlanAction(planId: string, billingPeriod: string): Promise<ActionResponse> {
      try {
        const api = await getServerApi();
        await api.client.POST("/api/v1/billing/subscription/change" as any, {
          body: { plan_id: planId, billing_period: billingPeriod } as any,
        });
        revalidatePath("/billing");
        return { status: "success", message: "Plan updated." };
      } catch (error) { return handleActionError(error, "Unable to change plan."); }
    }

    export async function cancelSubscriptionAction(): Promise<ActionResponse> { ... }
    export async function reactivateSubscriptionAction(): Promise<ActionResponse> { ... }
    export async function createPortalSessionAction(): Promise<ActionResponse & { portal_url?: string }> { ... }
    export async function previewProrationAction(planId: string, billingPeriod: string): Promise<ActionResponse & { preview?: ProrationPreviewOut }> { ... }
    ```

    Each mutation (change, cancel, reactivate) calls revalidatePath('/billing').
    createPortalSessionAction calls POST /api/v1/billing/portal-session, returns portal_url in response.
    previewProrationAction calls POST /api/v1/billing/proration-preview, returns preview data (NO revalidatePath — preview doesn't mutate state).

    Import ProrationPreviewOut from "@/lib/api/server".
