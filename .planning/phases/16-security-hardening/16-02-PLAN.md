"""Custom 429 handler for rate-limited requests.

Per RFC 6585, rate-limited responses use HTTP 429 (Too Many Requests)
with a Retry-After header indicating when the client may retry.

Wire via: RATELIMIT_VIEW = "apps.core.ratelimit.ratelimited_error" in settings.
"""

from django.http import HttpRequest, JsonResponse


def ratelimited_error(request: HttpRequest, exception: Exception | None = None) -> JsonResponse:
    """Return RFC 6585 compliant 429 response with Retry-After header.

    The Retry-After value is the rate window length in seconds.
    Default 60 seconds matches the '5/m' (per-minute) window.
    Projects using tighter windows (e.g., '5/s') should override this view
    or compute the value from RATELIMIT_AUTH_LOGIN dynamically.
    """
    response = JsonResponse(
        {
            "error": {
                "code": "RATE_001",
                "message": "Too many requests. Please try again later.",
            }
        },
        status=429,
    )
    response["Retry-After"] = "60"  # seconds — matches '5/m' window
    return response
```

# Auth endpoint URL prefixes subject to rate limiting.
# Covers both browser-session and app (JWT) allauth headless flows.
_AUTH_RATE_LIMIT_PATHS = (
    "/api/v1/auth/browser/v1/",
    "/api/v1/auth/app/v1/",
)


class AuthRateLimitMiddleware:
    """Rate-limit allauth headless auth endpoints via is_ratelimited() helper.

    Applied as middleware (not @ratelimit decorator) because allauth's headless
    views are registered internally and are not owned code.

    RATELIMIT_FAIL_OPEN = True means requests pass through if Redis is
    unreachable — auth availability is prioritised over strict enforcement
    during cache outages.

    Disabled in tests via conftest autouse fixture:
        settings.RATELIMIT_ENABLE = False
    """

    def __init__(self, get_response) -> None:
        self.get_response = get_response

    def __call__(self, request: HttpRequest) -> HttpResponse:
        if request.path.startswith(_AUTH_RATE_LIMIT_PATHS):
            from django.conf import settings

            rate = getattr(settings, "RATELIMIT_AUTH_LOGIN", "5/m")
            limited = is_ratelimited(
                request,
                group="auth",
                key="ip",
                rate=rate,
                increment=True,
            )
            if limited:
                raise Ratelimited()
        return self.get_response(request)
```

# --- SEC-01: Rate Limiting ---

def test_rate_limit_returns_429(client):
    """Auth endpoints return HTTP 429 after exceeding the configured rate limit."""
    # Enable rate limiting for this test (autouse fixture disables it globally).
    # Use 1/m so 2 requests trigger the limit without needing 5 real requests.
    with override_settings(RATELIMIT_ENABLE=True, RATELIMIT_AUTH_LOGIN="1/m"):
        # First request — should pass (within limit)
        client.post(
            "/api/v1/auth/browser/v1/auth/login",
            data={"username": "x", "password": "y"},
            content_type="application/json",
        )
        # Second request — should be rate limited (exceeds 1/m)
        response = client.post(
            "/api/v1/auth/browser/v1/auth/login",
            data={"username": "x", "password": "y"},
            content_type="application/json",
        )
    assert response.status_code == 429


def test_rate_limit_retry_after_header(client):
    """HTTP 429 response includes a Retry-After header."""
    with override_settings(RATELIMIT_ENABLE=True, RATELIMIT_AUTH_LOGIN="1/m"):
        client.post(
            "/api/v1/auth/browser/v1/auth/login",
            data={},
            content_type="application/json",
        )
        response = client.post(
            "/api/v1/auth/browser/v1/auth/login",
            data={},
            content_type="application/json",

    assert response.status_code == 429
    assert response.get("Retry-After") == "60"
