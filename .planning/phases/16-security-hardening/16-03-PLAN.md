```python
# django-csp 4.0 uses CONTENT_SECURITY_POLICY dict (NOT old CSP_* flat settings).
# Old API (CSP_DEFAULT_SRC etc.) is silently ignored in 4.0 — do not mix APIs.
#
# NONCE sentinel: django-csp replaces "NONCE" with 'nonce-{unique_value}' per request.
# request.csp_nonce is available in templates for inline scripts that must be whitelisted.
#
# Third-party sources:
#   PostHog: eu.i.posthog.com and us.i.posthog.com (EU/US data regions)
#   Stripe:  js.stripe.com (SDK), hooks.stripe.com (webhook ping JS)
#
# upgrade-insecure-requests omitted — HSTS in production.py handles upgrade-to-HTTPS.
# Adding it here would break local HTTP development.
CONTENT_SECURITY_POLICY = {
    "DIRECTIVES": {
        "default-src": ["'self'"],
        "script-src": ["'self'", "NONCE", "https://js.stripe.com"],
        "script-src-elem": ["'self'", "NONCE", "https://js.stripe.com"],
        "style-src": ["'self'", "NONCE"],
        "img-src": ["'self'", "data:", "blob:"],
        "font-src": ["'self'"],
        "connect-src": [
            "'self'",
            "https://eu.i.posthog.com",
            "https://us.i.posthog.com",
            "https://hooks.stripe.com",
        ],
        "frame-src": ["https://js.stripe.com"],
        "object-src": ["'none'"],
        "base-uri": ["'self'"],
        "form-action": ["'self'"],
        "frame-ancestors": ["'none'"],
    },
}

# X-Frame-Options: Django's XFrameOptionsMiddleware reads X_FRAME_OPTIONS.
# "DENY" is the Django default but we set it explicitly for auditability.
X_FRAME_OPTIONS = "DENY"

# Referrer-Policy: Django SecurityMiddleware reads SECURE_REFERRER_POLICY.
```

**Step 1 — Update apps/web/next.config.ts:**

Add the headers() async function to the existing nextConfig. Import nothing new — all values are inline:

```typescript
import path from "node:path";

import type { NextConfig } from "next";

const isDev = process.env.NODE_ENV === "development";

// Security headers applied to all Next.js routes.
// CSP here is a STATIC policy (no nonces) — Next.js App Router generates
// inline hydration scripts that require 'unsafe-inline' when nonces are not
// injected via Next.js middleware. Django's CSP uses nonces (via django-csp);
// these are separate origins. No conflict.
//
// isDev adds 'unsafe-eval' for Next.js hot-module reload in development.
// HSTS is intentionally omitted — CDN/reverse proxy handles HTTPS upgrade in prod.
const cspDirectives = [
  "default-src 'self'",
  isDev
    ? "script-src 'self' 'unsafe-inline' 'unsafe-eval' https://js.stripe.com"
    : "script-src 'self' 'unsafe-inline' https://js.stripe.com",
  "style-src 'self' 'unsafe-inline'",
  "img-src 'self' data: blob:",
  "font-src 'self'",
  "connect-src 'self' https://eu.i.posthog.com https://us.i.posthog.com https://hooks.stripe.com",
  "frame-src https://js.stripe.com",
  "object-src 'none'",
  "base-uri 'self'",
  "form-action 'self'",
  "frame-ancestors 'none'",
].join("; ");

const nextConfig: NextConfig = {
  // Enable standalone output for Docker deployments
  output: "standalone",

  async headers() {
    return [
      {
        source: "/(.*)",
        headers: [
          { key: "Content-Security-Policy", value: cspDirectives },
          { key: "X-Frame-Options", value: "DENY" },
          { key: "Referrer-Policy", value: "strict-origin-when-cross-origin" },
          { key: "X-Content-Type-Options", value: "nosniff" },
        ],
      },
    ];
  },

  // React Native-flavored packages ship untranspiled code, so let Next.js compile them.
  transpilePackages: ["@repo/ui", "@repo/business-logic", "@repo/validation"],
  webpack: (config) => {
    config.resolve = config.resolve ?? {};
    config.resolve.alias = {
      ...(config.resolve.alias ?? {}),
      "react-native$": "react-native-web",
      "@repo/validation": path.resolve(__dirname, "../../packages/validation"),
    };
    const extensions = config.resolve.extensions ?? [".tsx", ".ts", ".js"];
    config.resolve.extensions = Array.from(
      new Set([
        ...extensions,
        ".web.tsx",
        ".web.ts",
        ".web.js",
        ".native.tsx",
        ".native.ts",
        ".native.js",
      ]),
    );
    return config;
  },
};

export default nextConfig;
```

```python
# --- SEC-02: Security Headers ---

def test_csp_header_present(client):
    """Every Django response includes a Content-Security-Policy header."""
    response = client.get("/api/health/live")  # health endpoint — no auth needed
    assert "Content-Security-Policy" in response, (
        "Content-Security-Policy header missing — is csp.middleware.CSPMiddleware in MIDDLEWARE?"
    )
    assert "default-src" in response["Content-Security-Policy"]


def test_hsts_header_production(client):
    """Production settings include Strict-Transport-Security header."""
    # HSTS is in production.py only. Simulate production by overriding settings.
    # SECURE_SSL_REDIRECT=False prevents Django from redirecting the test client to HTTPS.
    with override_settings(SECURE_HSTS_SECONDS=31536000, SECURE_SSL_REDIRECT=False):
        response = client.get("/api/health/live")
    assert "Strict-Transport-Security" in response, (
        "Strict-Transport-Security header missing — is SECURE_HSTS_SECONDS set in production.py?"
    )
    assert "max-age=31536000" in response["Strict-Transport-Security"]
    assert "includeSubDomains" in response["Strict-Transport-Security"]


def test_x_frame_options_header(client):
    """Every Django response includes X-Frame-Options: DENY."""
    response = client.get("/api/health/live")
    assert response.get("X-Frame-Options") == "DENY", (
        f"Expected X-Frame-Options: DENY, got: {response.get('X-Frame-Options')!r}"
    )


def test_referrer_policy_header(client):
    """Every Django response includes Referrer-Policy: strict-origin-when-cross-origin."""
    response = client.get("/api/health/live")
    assert response.get("Referrer-Policy") == "strict-origin-when-cross-origin", (
        f"Expected Referrer-Policy: strict-origin-when-cross-origin, got: {response.get('Referrer-Policy')!r}"
    )
```

Note: Tests use `/api/health/live` because it is a known-working endpoint with no auth requirement (established in Phase 15). If health endpoint path differs, check `apps/api/apps/core/health_urls.py` for the correct path and adjust.
