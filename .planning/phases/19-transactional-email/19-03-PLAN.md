```python
# apps/api/apps/notifications/utils.py
import structlog
from django.db import transaction

logger = structlog.get_logger(**name**)

def send_email(
to: str,
template_name: str,
context: dict,
\*,
from_email: str | None = None,
reply_to: str | None = None,
) -> None:
"""Queue a transactional email via Celery after the current transaction commits.

    Email is never sent if the triggering transaction rolls back.
    All context dict values must be JSON-serializable primitives (str, int, bool, list, dict).
    """
    from apps.notifications.tasks import send_email_task

    def enqueue() -> None:
        send_email_task.delay(
            to=to,
            template_name=template_name,
            context=context,
            from_email=from_email,
            reply_to=reply_to,
        )
        logger.info("email_queued", to=to, template_name=template_name)

    transaction.on_commit(enqueue)

````

```python
def _extract_subject(full_html: str) -> tuple[str, str]:
    first_line, _, body = full_html.partition("\n")
    # first_line format: "<!-- SUBJECT: Welcome to the app -->"
    subject = first_line.replace("<!-- SUBJECT:", "").replace("-->", "").strip()
    return subject, body
```
