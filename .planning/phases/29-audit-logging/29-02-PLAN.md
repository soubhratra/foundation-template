AuditContextMiddleware must subclass pghistory.middleware.HistoryMiddleware:

```python
import pghistory.middleware

class AuditContextMiddleware(pghistory.middleware.HistoryMiddleware):
    def get_context(self, request):
        ctx = super().get_context(request)
        org = getattr(request, "org", None)
        if org is not None:
            ctx["org_id"] = str(org.id)
        return ctx
```

audit_context() must check INSTALLED_APPS and degrade gracefully:

```python
from contextlib import contextmanager, nullcontext

@contextmanager
def audit_context(user_id=None, org_id=None):
    from django.apps import apps as django_apps
    if not django_apps.is_installed("pghistory"):
        with nullcontext():
            yield
        return
    import pghistory
    with pghistory.context(
        user=user_id,
        org_id=str(org_id) if org_id is not None else None,
    ):
        yield
```
