```python
"""Schemas for the audit log read API.

AuditEntryOut intentionally omits the 'impersonator' field.
That field is stored in pgh_context metadata by Phase 30 (django-hijack
signal handler), but is staff-only and must not be exposed to org-level
queries. The exclusion is structural: the field is not in this schema.
"""
from __future__ import annotations

from datetime import datetime
from typing import Any

from ninja import Schema


class AuditEntryOut(Schema):
    pgh_id: int
    pgh_created_at: datetime
    pgh_label: str          # e.g. "org.updated", "api_key.created"
    pgh_obj_model: str      # e.g. "orgs.Organization"
    pgh_obj_id: str         # stringified PK of the changed object
    actor_id: int | None    # from pgh_context metadata "user" key; None for system changes
    url: str | None         # from pgh_context metadata "url" key; None for task/system changes
    # impersonator intentionally absent — staff-only, added in Phase 30 admin view

    @classmethod
    def from_event(cls, event: Any) -> "AuditEntryOut":
        """Construct from a pghistory Events CTE row.

        pgh_context is None when the change happened outside a request
        (e.g. management commands, test fixtures without pghistory.context()).
        Always guard with 'or {}'.
        """
        metadata: dict = getattr(event, "pgh_context", None) or {}
        return cls(
            pgh_id=event.pgh_id,
            pgh_created_at=event.pgh_created_at,
            pgh_label=event.pgh_label,
            pgh_obj_model=event.pgh_obj_model,
            pgh_obj_id=str(event.pgh_obj_id),
            actor_id=metadata.get("user"),
            url=metadata.get("url"),
        )


class AuditLogResponse(Schema):
    count: int
    results: list[AuditEntryOut]
```

Create `apps/api/apps/audit/controllers.py`:

The endpoint must use session auth AND JWT auth (same pattern as other B2B controllers). Permission gate is IsOrgAdmin — this permission class also sets request.org on the request, which AuditContextMiddleware then reads.

The org-scoping query has two parts joined with a UNION workaround (Pitfall 3 from research):

1. `Events.objects.references(org)` — covers Organization, Membership, WebhookEndpoint events (all have direct org FK)
2. `Events.objects.across("api_keys.APIKeyEvent").filter(pgh_data__user__in=org_user_ids)` — covers APIKey events for org members (APIKey has user FK only, no org FK)

IMPORTANT: pghistory.models.Events returns a special CTE queryset. Standard Django UNION (`|`) may not work. Instead, fetch both querysets, combine as Python lists for pagination, or use a single `.filter()` with Q objects if the CTE supports it. Simpler approach: perform both queries and concatenate results in Python before slicing for pagination. This avoids ORM UNION complexity with CTEs.

Recommended implementation:

```python
"""AuditLogController: read-only org audit log API."""
from __future__ import annotations  # remove if it breaks Ninja schema resolution

import pghistory.models
from ninja_extra import ControllerBase, api_controller, route
from ninja_extra.permissions import IsAuthenticated

from apps.audit.schemas import AuditEntryOut, AuditLogResponse
from apps.core.auth import SessionAuth, jwt_token_auth
from apps.orgs.permissions import IsOrgAdmin


@api_controller(
    "/org/{org_slug}/audit-log",
    tags=["audit"],
    auth=[SessionAuth(), jwt_token_auth],
    permissions=[IsAuthenticated, IsOrgAdmin],
)
class AuditLogController(ControllerBase):
    @route.get(
        "/",
        response={200: AuditLogResponse},
        url_name="audit-log-list",
    )
    def list_audit_log(
        self,
        request,
        org_slug: str,
        page: int = 1,
        page_size: int = 50,
        actor: int | None = None,
        action: str | None = None,
        date_from: str | None = None,
        date_to: str | None = None,
    ):
        org = request.org  # set by IsOrgAdmin

        # Part 1: events for models with direct org FK
        # (Organization, Membership, WebhookEndpoint)
        qs_org = pghistory.models.Events.objects.references(org).order_by(
            "-pgh_created_at"
        )

        # Part 2: APIKey events — APIKey has user FK, not org FK
        # Scope to users who are members of this org
        org_user_ids = list(org.memberships.values_list("user_id", flat=True))
        qs_keys = (
            pghistory.models.Events.objects.across("api_keys.APIKeyEvent")
            .filter(pgh_data__user__in=org_user_ids)
            .order_by("-pgh_created_at")
        )

        # Apply shared filters to both querysets
        if actor is not None:
            qs_org = qs_org.filter(pgh_context__metadata__user=actor)
            qs_keys = qs_keys.filter(pgh_context__metadata__user=actor)
        if action is not None:
            qs_org = qs_org.filter(pgh_label=action)
            qs_keys = qs_keys.filter(pgh_label=action)
        if date_from:
            qs_org = qs_org.filter(pgh_created_at__date__gte=date_from)
            qs_keys = qs_keys.filter(pgh_created_at__date__gte=date_from)
        if date_to:
            qs_org = qs_org.filter(pgh_created_at__date__lte=date_to)
            qs_keys = qs_keys.filter(pgh_created_at__date__lte=date_to)

        # Combine, sort by newest first, paginate in Python
        # Note: CTE-based querysets do not support SQL UNION directly
        all_events = sorted(
            list(qs_org) + list(qs_keys),
            key=lambda e: e.pgh_created_at,
            reverse=True,
        )
        count = len(all_events)
        offset = (page - 1) * page_size
        page_events = all_events[offset : offset + page_size]

        return 200, AuditLogResponse(
            count=count,
            results=[AuditEntryOut.from_event(e) for e in page_events],
        )
```
