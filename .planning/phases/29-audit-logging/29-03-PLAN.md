Cursor encode/decode on (pgh_created_at, pgh_id):

```python
import base64
from datetime import datetime

def encode_cursor(ts: datetime, pgh_id: int) -> str:
    raw = f"{ts.isoformat()}:{pgh_id}"
    return base64.urlsafe_b64encode(raw.encode()).decode()

def decode_cursor(cursor: str) -> tuple[datetime, int] | None:
    try:
        raw = base64.urlsafe_b64decode(cursor.encode()).decode()
        ts_str, id_str = raw.rsplit(":", 1)
        return datetime.fromisoformat(ts_str), int(id_str)
    except Exception:
        return None
```

"""Audit log response schemas.

IMPORTANT: No `from __future__ import annotations` — breaks Pydantic QueryParams
in django-ninja-extra (confirmed Phase 27 anti-pattern).
"""
from datetime import datetime
from typing import Optional
from ninja import Schema
from ninja_extra import QueryParams


class AuditEntryOut(Schema):
    id: int                     # pgh_id — auto-increment within event table
    timestamp: datetime         # pgh_created_at (serialised as ISO 8601 UTC)
    action: str                 # pgh_label e.g. "org.updated"
    model_name: str             # e.g. "Organization"
    object_id: str              # pgh_obj_id as string
    actor_id: Optional[str]     # user pk from pgh_context["user"], None if deleted
    actor_email: Optional[str]  # resolved via bulk User lookup, None if deleted
    url: Optional[str]          # from pgh_context["url"]
    # impersonator intentionally omitted — staff-only; populated in Phase 30


class CursorPageOut(Schema):
    items: list[AuditEntryOut]
    next_cursor: Optional[str]
    page_size: int


class AuditLogQueryParams(QueryParams):
    actor_id: Optional[int] = None   # user pk (int, not UUID — pghistory stores pk as BigAutoField int)
    action: Optional[str] = None     # exact match e.g. "org.updated"
    from_ts: Optional[datetime] = None   # ISO 8601, maps to pgh_created_at__gte
    to_ts: Optional[datetime] = None     # ISO 8601, maps to pgh_created_at__lte
    cursor: Optional[str] = None
    page_size: int = 50
```

"""AuditLogController — GET /api/v1/org/{slug}/audit-log

IMPORTANT: No `from __future__ import annotations` — breaks Pydantic QueryParams (Phase 27 anti-pattern).
"""
from typing import Any
from django.http import HttpRequest
from ninja import Query
from ninja_extra import api_controller, route, ControllerBase
from ninja_extra.permissions import IsAuthenticated

from apps.orgs.permissions import IsOrgMember, IsOrgAdmin
from apps.audit.schemas import AuditLogQueryParams, CursorPageOut
from apps.audit.selectors import get_audit_log_page


@api_controller(
    "/org/{org_slug}/audit-log",
    tags=["Audit Log"],
    permissions=[IsAuthenticated, IsOrgMember, IsOrgAdmin],
)
class AuditLogController(ControllerBase):
    @route.get("", response=CursorPageOut)
    def get_audit_log(
        self,
        request: HttpRequest,
        org_slug: str,
        filters: AuditLogQueryParams = Query(...),
    ) -> Any:
        """Return paginated audit log entries for this organization.

        Org owners and admins only. Impersonator field is never serialised
        (stored in pghistory context metadata, excluded from AuditEntryOut).
        """
        page_size = min(filters.page_size, 200)
        items, next_cursor = get_audit_log_page(
            org=request.org,
            actor_id=filters.actor_id,
            from_ts=filters.from_ts,
            to_ts=filters.to_ts,
            action=filters.action,
            cursor=filters.cursor,
            page_size=page_size,
        )
        return {"items": items, "next_cursor": next_cursor, "page_size": page_size}
```

Note: auto_discover_controllers() in config/urls.py will pick up AuditLogController automatically because apps.audit is in INSTALLED_APPS. No manual URL wiring needed.
