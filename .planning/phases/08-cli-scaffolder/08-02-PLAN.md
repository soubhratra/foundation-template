           '**/pyproject.toml',
           '**/*.md',
           '**/*.ts',
           '**/*.tsx',
           '**/*.py',
           '**/*.yml',
           '**/*.yaml',
           '**/.env.example'
         ], {
           cwd: outputDir,
           absolute: true,
           ignore: ['**/node_modules/**', '**/dist/**', '**/.venv/**']

         - Read content as UTF-8
         - Apply all replacements using regex with global flag (use helper to escape regex special chars)
