```
POST   /api/v1/org/                                    # create org (IsAuthenticated)
```

    ```python
    import re
    from datetime import datetime
    from ninja import Schema
    from pydantic import field_validator

    class OrgIn(Schema):
        name: str
        slug: str

        @field_validator("slug")
        @classmethod
        def slug_must_be_lowercase_alphanumeric(cls, v: str) -> str:
            if not re.match(r"^[a-z0-9-]+$", v):
                raise ValueError("Slug must contain only lowercase letters, numbers, and hyphens")
            return v.lower()

    class OrgOut(Schema):
        id: int
        name: str
        slug: str
        created_at: datetime

    class OrgUpdateIn(Schema):
        name: str | None = None

    class MemberOut(Schema):
        user_id: int
        email: str
        role: str
        joined_at: datetime  # maps to Membership.created_at

    class RoleUpdateIn(Schema):
        role: str  # "admin" or "member" — validated in service layer

    class TransferOwnershipIn(Schema):
        new_owner_user_id: int

    class InviteIn(Schema):
        email: str

    class InviteOut(Schema):
        id: int
        invited_email: str
        status: str
        expires_at: datetime
        created_at: datetime

    class AcceptInviteOut(Schema):
        redirect: str | None = None
        message: str | None = None
    ```

    Note: MemberOut.email maps from `membership.user.email` — the controller resolves this by reading `membership.user` (possible because services use `select_related("user")`). MemberOut.joined_at maps from `membership.created_at`. The schema uses `from_orm` mode — ensure all fields are accessible as attributes.
