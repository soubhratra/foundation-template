```python
from django.db import transaction

def transfer_ownership(self, org, current_owner, new_owner_user) -> None:
    with transaction.atomic():
        Membership.objects.filter(org=org, user=current_owner).update(role=Membership.Role.ADMIN)
        Membership.objects.filter(org=org, user=new_owner_user).update(role=Membership.Role.OWNER)
```

    ```python
    def _resolve_org_plan_id(org) -> str | None:
        """Return the active plan_id for an org's subscription, or None if no active subscription."""
        try:
            from apps.orgs.models import OrgCustomer
            from apps.billing.models import Subscription
            org_customer = OrgCustomer.objects.get(org=org)
            sub = Subscription.objects.filter(
                customer__stripe_customer_id=org_customer.stripe_customer_id,
                status="active",
            ).first()
            if sub and sub.plan_id:
                return sub.plan_id
        except Exception:
            pass
        return None
    ```
