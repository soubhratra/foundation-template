<objective>
Create the missing `(auth)/error/page.tsx` that handles OAuth error redirects. When an allauth OAuth flow fails (user cancels, provider rejects, unexpected error), allauth redirects to `HEADLESS_FRONTEND_URLS["socialaccount_login_error"]` which is `/auth/error?error=<code>`. This page must read the `?error=` query param, map it to a human-readable message, and display it with a link back to sign-in.

Purpose: Without this page, any OAuth failure (including user-initiated cancellation) ends in a 404, making error recovery impossible.
Output: `apps/web/app/(auth)/error/page.tsx` (working error page) and `apps/web/src/__tests__/app/(auth)/error.test.tsx` (RED tests turned GREEN).

```typescript
"use client";

import { useSearchParams } from "next/navigation";
import { AuthCard, AuthLinks } from "@/components/auth";

const ERROR_MESSAGES: Record<string, string> = {
  unknown: "An unexpected error occurred. Please try again.",
  cancelled: "You cancelled the sign-in. Try again when ready.",
  denied: "Access was denied by the provider. Please try again.",
  reauthentication_required: "Re-authentication required. Please sign in again.",
  signup_closed: "Account registration is currently closed.",
  permission_denied: "You do not have permission to sign in.",
};

export default function OAuthErrorPage() {
  const searchParams = useSearchParams();
  const errorCode = searchParams?.get("error") ?? "unknown";
  const message = ERROR_MESSAGES[errorCode] ?? ERROR_MESSAGES.unknown;

  return (
    <AuthCard
      eyebrow="Authentication error"
      title="Sign-in failed"
      description={message}
      footer={
        <AuthLinks
          items={[
            {
              prefix: "Ready to try again?",
              label: "Back to sign-in",
              href: "/",
            },
          ]}
        />
      }
    >
      <p className="text-center text-sm text-muted-foreground">
        Error code: <code className="font-mono">{errorCode}</code>
      </p>
    </AuthCard>
  );
}
```

Do NOT use a switch/case — the `Record<string, string>` lookup is sufficient for the fixed allauth error code set and is easier to extend.
Do NOT use `AuthCard` without the `footer` prop — the back-to-sign-in link is required for error recovery.
The `(auth)/layout.tsx` already exports `dynamic = "force-dynamic"` — no additional Suspense wrapper needed for `useSearchParams`.
