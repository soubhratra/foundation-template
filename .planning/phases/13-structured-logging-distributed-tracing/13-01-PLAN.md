    ```python
    """Tests for OBS-01: structlog processor and renderer selection."""
    import pytest
    from unittest.mock import MagicMock, patch


    def test_otel_processor_injects_when_span_active():
        """add_otel_context injects trace_id + span_id when a span is recording."""
        from apps.core.logging import add_otel_context

        mock_ctx = MagicMock()
        mock_ctx.trace_id = 0xABCDEF1234567890ABCDEF1234567890
        mock_ctx.span_id = 0xABCDEF1234567890

        mock_span = MagicMock()
        mock_span.is_recording.return_value = True
        mock_span.get_span_context.return_value = mock_ctx

        with patch("opentelemetry.trace.get_current_span", return_value=mock_span):
            event_dict = {}
            result = add_otel_context(None, None, event_dict)

        assert "trace_id" in result
        assert "span_id" in result
        assert len(result["trace_id"]) == 32
        assert len(result["span_id"]) == 16
        assert result["trace_id"] == format(mock_ctx.trace_id, "032x")
        assert result["span_id"] == format(mock_ctx.span_id, "016x")


    def test_otel_processor_omits_when_no_span():
        """add_otel_context omits trace_id + span_id (not null) when no span is active."""
        from apps.core.logging import add_otel_context
        from opentelemetry.trace import NonRecordingSpan, INVALID_SPAN_CONTEXT

        non_recording = NonRecordingSpan(INVALID_SPAN_CONTEXT)

        with patch("opentelemetry.trace.get_current_span", return_value=non_recording):
            event_dict = {}
            result = add_otel_context(None, None, event_dict)

        assert "trace_id" not in result
        assert "span_id" not in result


    def test_renderer_selection_dev(settings):
        """ConsoleRenderer is active in dev (DEBUG=True)."""
        import structlog
        settings.DEBUG = True
        from apps.core.logging import configure_structlog
        configure_structlog()
        # structlog is configured — no exception means success
        assert structlog.is_configured()


    def test_renderer_selection_prod(settings):
        """JSONRenderer is active in prod (DEBUG=False)."""
        import structlog
        settings.DEBUG = False
        from apps.core.logging import configure_structlog
        configure_structlog()
        assert structlog.is_configured()
    ```

    ```python
    """Tests for OBS-02: configure_otel() no-op and active TracerProvider."""
    import os
    import pytest
    from unittest.mock import patch
    from opentelemetry import trace
    from opentelemetry.sdk.trace import TracerProvider


    def test_configure_otel_noop_when_no_env(monkeypatch):
        """configure_otel() is a no-op when OTEL_EXPORTER_OTLP_ENDPOINT is not set."""
        from apps.core.tracing import configure_otel

        monkeypatch.delenv("OTEL_EXPORTER_OTLP_ENDPOINT", raising=False)
        # Reset to default no-op provider first
        from opentelemetry.sdk.trace import TracerProvider as _TP
        # Record pre-call state — default provider is ProxyTracerProvider, not SDK TracerProvider
        configure_otel()
        # After no-op call, provider must still NOT be a real TracerProvider
        provider = trace.get_tracer_provider()
        # The no-op case: provider is still the global proxy (not SDK TracerProvider)
        # We verify by checking it can't export (no processors attribute)
        assert not isinstance(provider, TracerProvider), (
            "configure_otel() should NOT set a real TracerProvider when endpoint is absent"
        )


    def test_configure_otel_active_when_env_set(monkeypatch):
        """configure_otel() creates a real TracerProvider when endpoint is set."""
        from apps.core.tracing import configure_otel

        monkeypatch.setenv("OTEL_EXPORTER_OTLP_ENDPOINT", "http://localhost:4317")
        # Patch OTLPSpanExporter to avoid actual network connection
        with patch("opentelemetry.exporter.otlp.proto.grpc.trace_exporter.OTLPSpanExporter") as MockExporter:
            MockExporter.return_value = MockExporter
            configure_otel()

        provider = trace.get_tracer_provider()
        assert isinstance(provider, TracerProvider), (
            "configure_otel() must set a real SDK TracerProvider when endpoint is set"
        )
        # Cleanup: reset global provider to avoid test pollution
        from opentelemetry import trace as _trace
        _trace._TRACER_PROVIDER = None
    ```
