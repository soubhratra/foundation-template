    2. Create `apps/api/apps/core/tracing.py`:
    ```python
    """OpenTelemetry configuration helper for Phase 13.

    configure_otel() is designed to be called:
    - At Django startup via CoreConfig.ready() (for the web process)
    - Inside @worker_process_init.connect in config/celery.py (for Celery workers, after fork)

    Design: zero-overhead no-op when OTEL_EXPORTER_OTLP_ENDPOINT is absent.
    OTLPSpanExporter is imported lazily to avoid ~100ms import cost when tracing is disabled.
    insecure=True is correct for Jaeger all-in-one on port 4317 (no TLS in Docker Compose).
    """
    from __future__ import annotations

    import os

    from opentelemetry import trace
    from opentelemetry.sdk.resources import SERVICE_NAME, Resource
    from opentelemetry.sdk.trace import TracerProvider
    from opentelemetry.sdk.trace.export import BatchSpanProcessor


    def configure_otel(service_name: str | None = None) -> None:
        """Configure OpenTelemetry if OTEL_EXPORTER_OTLP_ENDPOINT is set.

        No-op (and no import of OTLPSpanExporter) when endpoint is absent, so CI
        and local dev without Jaeger incur zero overhead and no failed connections.

        Args:
            service_name: Override service name in Jaeger. Falls back to
                OTEL_SERVICE_NAME env var, then DJANGO_SETTINGS_MODULE value,
                then the literal string "foundation-api".
        """
        endpoint = os.environ.get("OTEL_EXPORTER_OTLP_ENDPOINT", "")
        if not endpoint:
            return  # Remain on default no-op ProxyTracerProvider

        # Lazy import: only pay the import cost when tracing is actually enabled.
        from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter

        _service_name = service_name or os.environ.get(
            "OTEL_SERVICE_NAME",
            os.environ.get("DJANGO_SETTINGS_MODULE", "foundation-api"),
        )
        resource = Resource(attributes={SERVICE_NAME: _service_name})
        provider = TracerProvider(resource=resource)
        exporter = OTLPSpanExporter(endpoint=endpoint, insecure=True)
        provider.add_span_processor(BatchSpanProcessor(exporter))
        trace.set_tracer_provider(provider)
    ```

    ```python
    from django.apps import AppConfig


    class CoreConfig(AppConfig):
        default_auto_field = "django.db.models.BigAutoField"
        name = "apps.core"

        def ready(self) -> None:
            from apps.core.logging import configure_structlog
            from apps.core.tracing import configure_otel
            from opentelemetry.instrumentation.django import DjangoInstrumentor
            from opentelemetry.instrumentation.psycopg import PsycopgInstrumentor

            configure_structlog()
            configure_otel()

            # Auto-instrument Django views (creates spans for each request)
            # and psycopg3 SQL queries (creates child spans for each DB query).
            # Both are safe to call with the default no-op ProxyTracerProvider
            # when OTEL_EXPORTER_OTLP_ENDPOINT is absent — they produce no-op spans.
            DjangoInstrumentor().instrument()
            PsycopgInstrumentor().instrument()
    ```

    2. Update `apps/api/config/celery.py` — add DjangoStructLogInitStep and OTel worker init:

    ```python
    """Celery application configuration.

    Registers DjangoStructLogInitStep for structlog context propagation across
    Django request → Celery task boundaries (Phase 13).

    OTel tracing for Celery workers is initialized in init_celery_tracing() via
    worker_process_init signal (after prefork) to avoid losing BatchSpanProcessor
    threads across fork boundaries.
    """
    import os

    from celery import Celery
    from celery.signals import worker_process_init
    from django_structlog.celery.steps import DjangoStructLogInitStep

    # Set default Django settings module for 'celery' program
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "config.settings.development")

    app = Celery("foundation")

    # Load configuration from Django settings using CELERY_ namespace
    app.config_from_object("django.conf:settings", namespace="CELERY")

    # Autodiscover tasks from all installed apps
    app.autodiscover_tasks()

    # Register structlog context propagation step.
    # Handles task_prerun / task_postrun signals that bind task_id + task_name
    # and propagate request_id from the dispatching Django request into the task context.
    app.steps["worker"].add(DjangoStructLogInitStep)


    @worker_process_init.connect(weak=False)
    def init_celery_tracing(*args, **kwargs):
        """Initialize OTel tracing in each Celery worker process after prefork.

        MUST run inside worker_process_init (after fork). Celery uses a prefork model;
        BatchSpanProcessor spawns daemon threads that are lost across fork boundaries.
        Initializing here ensures each worker process has its own live processor threads.
        """
        from apps.core.tracing import configure_otel
        from opentelemetry.instrumentation.celery import CeleryInstrumentor

        configure_otel(service_name="foundation-celery-worker")
        CeleryInstrumentor().instrument()


    @app.task(bind=True, ignore_result=True)
    def debug_task(self):
        """Debug task to verify Celery is working."""
        print(f"Request: {self.request!r}")
    ```
