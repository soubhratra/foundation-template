```python
@contextmanager
def audit_context(user_id=None, org_id=None):
    """Wraps pghistory.context(). Degrades to no-op when pghistory not installed."""
    from django.apps import apps as django_apps
    if not django_apps.is_installed("pghistory"):
        with nullcontext():
            yield
        return
    import pghistory
    with pghistory.context(user=user_id, org_id=str(org_id) if org_id is not None else None):
        yield
```
