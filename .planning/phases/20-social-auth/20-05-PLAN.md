    2. Add handleAppleSignIn function inside the component (near the existing form handlers):
       ```typescript
       const handleAppleSignIn = useCallback(async () => {
         try {
           const credential = await AppleAuthentication.signInAsync({
             requestedScopes: [
               AppleAuthentication.AppleAuthenticationScope.FULL_NAME,
               AppleAuthentication.AppleAuthenticationScope.EMAIL,
             ],
           });

           if (!credential.identityToken) return;

           const response = await fetch(
             `${process.env.EXPO_PUBLIC_API_URL}/api/v1/auth/apple/`,
             {
               method: "POST",
               headers: { "Content-Type": "application/json" },
               body: JSON.stringify({
                 identity_token: credential.identityToken,
                 full_name: credential.fullName?.givenName ?? null,
               }),
             }
           );

           if (response.ok) {
             const tokens = await response.json();
             setTokens({ access: tokens.access, refresh: tokens.refresh });
             router.replace("/(app)");
           } else {
             setStatus({ type: "error", message: "Apple Sign In failed. Please try again." });
           }
         } catch (error: unknown) {
           // ERR_CANCELED: user dismissed the Apple sheet — silent dismiss
           if (
             error instanceof Error &&
             error.message.includes("ERR_CANCELED")
           ) {
             return;
           }
           setStatus({ type: "error", message: "Apple Sign In failed. Please try again." });
         }
       }, [router, setTokens]);
       ```
