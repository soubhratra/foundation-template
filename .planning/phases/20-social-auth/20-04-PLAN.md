    handleProviderRedirect(provider: "google" | "github"):
    ```typescript
    async function handleProviderRedirect(provider: "google" | "github") {
      const csrfToken = document.cookie
        .split("; ")
        .find((c) => c.startsWith("csrftoken="))
        ?.split("=")[1] ?? "";
      const response = await fetch("/api/v1/auth/browser/v1/auth/provider/redirect", {
        method: "POST",
        headers: { "Content-Type": "application/json", "X-CSRFToken": csrfToken },
        credentials: "include",
        body: JSON.stringify({ provider, process: "login", callback_url: "/auth/callback" }),
      });
      if (response.ok) {
        const data = await response.json();
        window.location.href = data.data.location;
      }
    }
    ```

    Error handling: if response not ok, show a sonner toast: `toast.error("Social login unavailable. Please try again.")`. Import toast from "@/components/ui/sonner".
