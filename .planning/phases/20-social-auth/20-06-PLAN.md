export async function removeEmailModule(outputDir: string): Promise<void> {
  // 1. Delete app directory (ENOENT guard)
  await rm(path.join(outputDir, "apps/api/apps/notifications"), { recursive: true, force: true });

  // 2. Strip sentinel-commented INSTALLED_APPS lines
  const basePyPath = path.join(outputDir, "apps/api/config/settings/base.py");
  let basePy = await fs.readFile(basePyPath, "utf-8");
  // Remove lines with # EMAIL MODULE inline comment
  basePy = basePy.replace(/^.*# EMAIL MODULE.*$/gm, "").replace(/\n{3,}/g, "\n\n");

  // 3. Remove fenced block
  basePy = basePy.replace(
    /\n# -{10,}\n# EMAIL MODULE.*?\n# -{10,}[\s\S]*?# -{10,}\n# END EMAIL MODULE\n# -{10,}/,
    "",
  );
  await fs.writeFile(basePyPath, basePy, "utf-8");

  // 4. Remove env vars from .env.example (ENOENT guard)
  const envExamplePath = path.join(outputDir, "apps/api/.env.example");
  try {
    let envExample = await fs.readFile(envExamplePath, "utf-8");
    envExample = envExample.replace(/^RESEND_API_KEY=.*$/m, "").replace(/\n{3,}/g, "\n\n");
    await fs.writeFile(envExamplePath, envExample, "utf-8");
  } catch (e) {
    if ((e as NodeJS.ErrnoException).code !== "ENOENT") throw e;
  }
}

    export async function removeSocialAuthModule(outputDir: string): Promise<void> {
      const basePyPath = path.join(outputDir, "apps/api/config/settings/base.py");

      // 1. Remove SOCIAL AUTH MODULE fenced block and sentinel-commented lines from base.py
      let basePy = await fs.readFile(basePyPath, "utf-8");

      // Remove the entire fenced block (# SOCIAL AUTH MODULE ... # END SOCIAL AUTH MODULE)
      basePy = basePy.replace(
        /\n# -{10,}\n# SOCIAL AUTH MODULE[^\n]*\n# -{10,}[\s\S]*?# -{10,}\n# END SOCIAL AUTH MODULE\n# -{10,}/,
        ""
      );

      // Remove any remaining lines with # SOCIAL AUTH MODULE inline comment
      // (handles any lines outside the fenced block, e.g., INSTALLED_APPS appends)
      basePy = basePy.replace(/^.*# SOCIAL AUTH MODULE.*$\n?/gm, "");

      // Remove allauth.socialaccount from INSTALLED_APPS (it was added in this module)
      basePy = basePy.replace(/^\s*"allauth\.socialaccount",?\s*# SOCIAL AUTH MODULE.*$\n?/gm, "");

      // Clean up excessive blank lines
      basePy = basePy.replace(/\n{3,}/g, "\n\n");
      await fs.writeFile(basePyPath, basePy, "utf-8");

      // 2. Remove social env vars from apps/api/.env.example (ENOENT guard)
      const apiEnvPath = path.join(outputDir, "apps/api/.env.example");
      try {
        let apiEnv = await fs.readFile(apiEnvPath, "utf-8");

          /^# -{10,}\n# Social Auth[\s\S]*?(?=\n# -{10,}|\z)/m,
          ""
