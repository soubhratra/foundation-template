    ```python
    from __future__ import annotations
    from datetime import datetime
    from typing import Optional
    from ninja import Schema
    from pydantic import field_validator


    # --- Endpoint schemas ---

    class EndpointIn(Schema):
        url: str
        name: str = ""
        event_types: list[str]

        @field_validator("url")
        @classmethod
        def url_must_be_https(cls, v: str) -> str:
            if not v.startswith("https://"):
                raise ValueError("Webhook endpoint URL must use HTTPS (http:// not allowed).")
            return v

        @field_validator("event_types")
        @classmethod
        def event_types_must_not_be_empty(cls, v: list[str]) -> list[str]:
            if not v:
                raise ValueError("event_types must contain at least one event type.")
            return v


    class EndpointOut(Schema):
        """Returned for list, get, update, delete — no secret field."""
        id: str
        url: str
        name: str
        event_types: list[str]
        is_active: bool
        consecutive_failures: int
        created_at: datetime

        @staticmethod
        def from_orm_endpoint(ep) -> "EndpointOut":
            return EndpointOut(
                id=str(ep.id),
                url=ep.url,
                name=ep.name,
                event_types=ep.event_types,
                is_active=ep.is_active,
                consecutive_failures=ep.consecutive_failures,
                created_at=ep.created_at,
            )


    class EndpointCreateOut(Schema):
        """Returned ONLY from POST /endpoints — includes secret shown once.
        secret == endpoint.signing_key (the raw HMAC key).
        """
        id: str
        url: str
        name: str
        event_types: list[str]
        is_active: bool
        created_at: datetime
        secret: str  # Raw signing_key — used by recipients for HMAC verification


    class EndpointUpdateIn(Schema):
        url: Optional[str] = None
        name: Optional[str] = None
        event_types: Optional[list[str]] = None
        is_active: Optional[bool] = None

        @field_validator("url")
        @classmethod
        def url_must_be_https(cls, v: Optional[str]) -> Optional[str]:
            if v is not None and not v.startswith("https://"):
                raise ValueError("Webhook endpoint URL must use HTTPS.")
            return v


    class RotateSecretOut(Schema):
        """Returned from POST /endpoints/{id}/rotate-secret — new secret shown once.
        secret == new endpoint.signing_key.
        """
        id: str
        secret: str  # New raw signing_key — not returned again


    # --- Delivery schemas ---

    class AttemptOut(Schema):
        attempt_number: int
        http_status_code: Optional[int]
        response_body: str
        duration_ms: Optional[int]
        attempted_at: datetime
        error_message: str


    class DeliveryOut(Schema):
        id: str
        event_type: str
        status: str
        created_at: datetime
        attempt_count: int  # Computed: delivery.attempts.count()

        @staticmethod
        def from_orm_delivery(d) -> "DeliveryOut":
            return DeliveryOut(
                id=str(d.id),
                event_type=d.event_type,
                status=d.status,
                created_at=d.created_at,
                attempt_count=d.attempt_count if hasattr(d, "attempt_count") else d.attempts.count(),
            )


    class DeliveryDetailOut(Schema):
        id: str
        event_type: str
        status: str
        request_body: dict
        created_at: datetime
        attempts: list[AttemptOut]
    ```

    ```python
    from __future__ import annotations

    from http import HTTPStatus
    from typing import Any

    from django.http import HttpRequest
    from django.shortcuts import get_object_or_404
    from ninja_extra import ControllerBase, api_controller, route
    from ninja_extra.permissions import IsAuthenticated
    from ninja.pagination import paginate, PageNumberPagination
    from django.db import models

    from apps.orgs.permissions import IsOrgAdmin, IsOrgMember
    from apps.webhooks.models import (
        DeliveryStatus,
        WebhookDelivery,
        WebhookEndpoint,
    )
    from apps.webhooks.schemas import (
        AttemptOut,
        DeliveryDetailOut,
        DeliveryOut,
        EndpointCreateOut,
        EndpointIn,
        EndpointOut,
        EndpointUpdateIn,
        RotateSecretOut,
    )
    from apps.webhooks.tasks import deliver_webhook


    @api_controller(
        "/org/{org_slug}/webhooks/endpoints",
        tags=["webhooks"],
        permissions=[IsAuthenticated, IsOrgMember],
    )
    class WebhookEndpointController(ControllerBase):

        @route.post("", response={201: EndpointCreateOut}, permissions=[IsAuthenticated, IsOrgAdmin])
        def create_endpoint(self, request: HttpRequest, org_slug: str, body: EndpointIn) -> Any:
            """Register a new webhook endpoint. Returns the raw secret (signing_key) exactly once."""
            raw_secret, signing_key, secret_hash = WebhookEndpoint.generate_secret()
            endpoint = WebhookEndpoint.objects.create(
                org=request.org,
                url=body.url,
                name=body.name,
                event_types=body.event_types,
                signing_key=signing_key,
                secret_hash=secret_hash,
            )
            return 201, EndpointCreateOut(
                id=str(endpoint.id),
                url=endpoint.url,
                name=endpoint.name,
                event_types=endpoint.event_types,
                is_active=endpoint.is_active,
                created_at=endpoint.created_at,
                secret=signing_key,  # signing_key == raw_secret; never returned again
            )

        @route.get("", response={200: list[EndpointOut]})
        def list_endpoints(self, request: HttpRequest, org_slug: str) -> Any:
            """List all webhook endpoints for the org (no secrets returned)."""
            endpoints = WebhookEndpoint.objects.filter(org=request.org).order_by("-created_at")
            return 200, [EndpointOut.from_orm_endpoint(ep) for ep in endpoints]

        @route.patch(
            "/{endpoint_id}",
            response={200: EndpointOut},
            permissions=[IsAuthenticated, IsOrgAdmin],
        )
        def update_endpoint(
            self, request: HttpRequest, org_slug: str, endpoint_id: str, body: EndpointUpdateIn
        ) -> Any:
            """Update endpoint fields. Re-enabling (is_active: true) resets consecutive_failures."""
            endpoint = get_object_or_404(WebhookEndpoint, id=endpoint_id, org=request.org)
            update_fields = []
            if body.url is not None:
                endpoint.url = body.url
                update_fields.append("url")
            if body.name is not None:
                endpoint.name = body.name
                update_fields.append("name")
            if body.event_types is not None:
                endpoint.event_types = body.event_types
                update_fields.append("event_types")
            if body.is_active is not None:
                endpoint.is_active = body.is_active
                update_fields.append("is_active")
                if body.is_active:
                    # Re-enabling resets circuit breaker counter per CONTEXT.md
                    endpoint.consecutive_failures = 0
                    update_fields.append("consecutive_failures")
            if update_fields:
                endpoint.save(update_fields=update_fields)
            return 200, EndpointOut.from_orm_endpoint(endpoint)

        @route.delete(
            "/{endpoint_id}",
            response={204: None},
            permissions=[IsAuthenticated, IsOrgAdmin],
        )
        def delete_endpoint(self, request: HttpRequest, org_slug: str, endpoint_id: str) -> Any:
            endpoint = get_object_or_404(WebhookEndpoint, id=endpoint_id, org=request.org)
            endpoint.delete()
            return 204, None

        @route.post(
            "/{endpoint_id}/rotate-secret",
            response={200: RotateSecretOut},
            permissions=[IsAuthenticated, IsOrgAdmin],
        )
        def rotate_secret(self, request: HttpRequest, org_slug: str, endpoint_id: str) -> Any:
            """Generate a new HMAC secret. Old secret is immediately invalidated."""
            endpoint = get_object_or_404(WebhookEndpoint, id=endpoint_id, org=request.org)
            raw_secret, signing_key, secret_hash = WebhookEndpoint.generate_secret()
            endpoint.signing_key = signing_key
            endpoint.secret_hash = secret_hash
            endpoint.save(update_fields=["signing_key", "secret_hash", "updated_at"])
            return 200, RotateSecretOut(id=str(endpoint.id), secret=signing_key)

        @route.get("/{endpoint_id}/deliveries", response={200: list[DeliveryOut]})
        @paginate(PageNumberPagination, page_size=20)
        def list_deliveries(self, request: HttpRequest, org_slug: str, endpoint_id: str) -> Any:
            """Paginated delivery history for an endpoint. Includes attempt_count per delivery."""
            endpoint = get_object_or_404(WebhookEndpoint, id=endpoint_id, org=request.org)
            return (
                WebhookDelivery.objects.filter(endpoint=endpoint)
                .annotate(attempt_count=models.Count("attempts"))
                .order_by("-created_at")
            )


    @api_controller(
        "/org/{org_slug}/webhooks/deliveries",
        tags=["webhooks"],
        permissions=[IsAuthenticated, IsOrgMember],
    )
    class WebhookDeliveryController(ControllerBase):

        @route.get("/{delivery_id}", response={200: DeliveryDetailOut})
        def get_delivery(self, request: HttpRequest, org_slug: str, delivery_id: str) -> Any:
            """Delivery detail with all attempts."""
            delivery = get_object_or_404(
                WebhookDelivery,
                id=delivery_id,
                endpoint__org=request.org,
            )
            attempts = [
                AttemptOut(
                    attempt_number=a.attempt_number,
                    http_status_code=a.http_status_code,
                    response_body=a.response_body,
                    duration_ms=a.duration_ms,
                    attempted_at=a.attempted_at,
                    error_message=a.error_message,
                )
                for a in delivery.attempts.all()
            ]
            return 200, DeliveryDetailOut(
                id=str(delivery.id),
                event_type=delivery.event_type,
                status=delivery.status,
                request_body=delivery.request_body,
                created_at=delivery.created_at,
                attempts=attempts,
            )

        @route.post(
            "/{delivery_id}/retry",
            response={202: dict},
            permissions=[IsAuthenticated, IsOrgMember],
        )
        def retry_delivery(self, request: HttpRequest, org_slug: str, delivery_id: str) -> Any:
            """Manually enqueue a retry for a failed delivery. Returns 202 immediately."""
            delivery = get_object_or_404(
                WebhookDelivery,
                id=delivery_id,
                endpoint__org=request.org,
            )
            deliver_webhook.delay(str(delivery.id))
            return 202, {"queued": True, "delivery_id": str(delivery.id)}
    ```
