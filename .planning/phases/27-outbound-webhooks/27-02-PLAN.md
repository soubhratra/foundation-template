    **models.py — WebhookEndpoint:**

    The secret design uses two fields:
    - `signing_key`: stores the raw 32-byte hex secret (used as HMAC-SHA256 key in delivery task)
    - `secret_hash`: stores SHA-256 hex of the raw secret (for secure lookup without exposing the raw value in logs)

    The `signing_key` is a random token, not a password — it IS the shared secret that recipients use to
    verify signatures. Storing it in plaintext is intentional and follows Stripe/GitHub convention where
    the shared secret must be known to both sides in identical form. `EndpointCreateOut.secret` returns
    `signing_key` once at registration.

    ```python
    import uuid
    import secrets
    import hashlib
    from django.db import models
    from django.conf import settings

    class WebhookEndpoint(models.Model):
        id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
        org = models.ForeignKey(
            "orgs.Organization",
            on_delete=models.CASCADE,
            related_name="webhook_endpoints",
        )
        url = models.URLField(max_length=2048)
        name = models.CharField(max_length=255, blank=True, default="")
        event_types = models.JSONField(default=list)
        signing_key = models.CharField(max_length=64)   # Raw secret — HMAC key for delivery signing
        secret_hash = models.CharField(max_length=64)   # SHA-256 hex of signing_key — for secure lookup
        is_active = models.BooleanField(default=True)
        consecutive_failures = models.IntegerField(default=0)
        created_at = models.DateTimeField(auto_now_add=True)
        updated_at = models.DateTimeField(auto_now=True)

        class Meta:
            db_table = "webhooks_endpoint"
            indexes = [
                models.Index(fields=["org", "is_active"]),
            ]

        def __str__(self) -> str:
            return f"{self.name or self.url} ({self.org_id})"

        @staticmethod
        def generate_secret() -> tuple[str, str, str]:
            """Returns (raw_secret, signing_key, secret_hash).

            signing_key == raw_secret (the value used as HMAC key and returned to org admin once).
            secret_hash == SHA-256 hex of raw_secret (stored for secure lookup).
            Store both fields; return signing_key as EndpointCreateOut.secret.
            """
            raw = secrets.token_hex(32)
            hashed = hashlib.sha256(raw.encode()).hexdigest()
            return raw, raw, hashed
    ```

    Mirror the send_email() pattern from apps/notifications/utils.py exactly.

    ```python
    """Public interface for firing outbound webhook events.

    fire_webhook_event() enqueues one Celery deliver_webhook task per matching active
    endpoint via transaction.on_commit() — no delivery if the calling transaction rolls back.
    """
    from __future__ import annotations
    import hashlib
    import hmac
    import time
    from typing import TYPE_CHECKING
    from django.conf import settings
    from django.core.exceptions import ValidationError
    from django.db import connection, transaction

    if TYPE_CHECKING:
        from apps.orgs.models import Organization


    def fire_webhook_event(org: "Organization", event_type: str, payload: dict) -> None:
        """Dispatch a webhook event to all active org endpoints subscribed to event_type.

        Validates event_type against WEBHOOK_EVENTS settings dict.
        Enqueues one Celery deliver_webhook task per matching active endpoint,
        dispatched via transaction.on_commit() so no delivery occurs if the
        triggering transaction rolls back.

        Raises ValueError if event_type is not in settings.WEBHOOK_EVENTS.
        """
        webhook_events: dict = getattr(settings, "WEBHOOK_EVENTS", {})
        if event_type not in webhook_events:
            raise ValueError(f"Unknown webhook event type: {event_type}")

        from apps.webhooks.models import WebhookDelivery, WebhookEndpoint
        from apps.webhooks.tasks import deliver_webhook

        # Build payload envelope — always includes webhook_version: "v1"
        envelope: dict = {
            "webhook_version": "v1",
            "event_type": event_type,
            "data": payload,
            "timestamp": int(time.time()),
        }

        endpoints = WebhookEndpoint.objects.filter(
            org=org,
            is_active=True,
        )
        # Filter in Python to check JSONField list membership
        matching = [ep for ep in endpoints if event_type in ep.event_types]

        for endpoint in matching:
            delivery = WebhookDelivery.objects.create(
                endpoint=endpoint,
                event_type=event_type,
                status="pending",
                request_body=envelope,
                request_headers={},
            )
            delivery_id = str(delivery.id)

            def enqueue(did: str = delivery_id) -> None:
                deliver_webhook.delay(did)

            if connection.in_atomic_block:
                transaction.on_commit(enqueue)
            else:
                enqueue()
    ```

    **checks.py — WEBHOOK_EVENTS system check:**
    ```python
    from django.conf import settings
    from django.core.checks import Warning, register

    @register()
    def check_webhook_events(app_configs, **kwargs):
        errors = []
        webhook_events = getattr(settings, "WEBHOOK_EVENTS", None)
        if webhook_events is None:
            errors.append(
                Warning(
                    "WEBHOOK_EVENTS setting is not defined.",
                    hint="Add WEBHOOK_EVENTS = {} to your settings or extend the default dict.",
                    id="webhooks.W001",
                )
            )
            return errors

        for event_type, config in webhook_events.items():
            if "description" not in config:
                errors.append(
                    Warning(
                        f"WEBHOOK_EVENTS['{event_type}'] is missing a 'description' key.",
                        hint="Add a human-readable description for each event type.",
                        id="webhooks.W002",
                    )
                )
            if "example_payload" not in config:
                errors.append(
                    Warning(
                        f"WEBHOOK_EVENTS['{event_type}'] is missing an 'example_payload' key.",
                        hint="Add an example_payload dict for each event type.",
                        id="webhooks.W003",
                    )
                )
        return errors
    ```
