    a) posthog.init() call at module level (outside any component), guarded by typeof window !== "undefined":
       ```typescript
       import posthog from "posthog-js";
       import { PostHogProvider, usePostHog } from "posthog-js/react";

       if (typeof window !== "undefined") {
         posthog.init(process.env.NEXT_PUBLIC_POSTHOG_KEY ?? "", {
           api_host: process.env.NEXT_PUBLIC_POSTHOG_HOST ?? "https://eu.i.posthog.com",
           capture_pageview: false,   // Manual via PostHogPageView
           capture_pageleave: true,
           session_recording: { maskAllInputs: true },
           loaded: (ph) => {
             if (!process.env.NEXT_PUBLIC_POSTHOG_KEY) ph.opt_out_capturing();
           },
         });
       }
       ```
       The loaded callback silences PostHog when no key is set (opts out), so builds without NEXT_PUBLIC_POSTHOG_KEY work cleanly.

    b) PostHogPageView inner component (must be INSIDE PostHogProvider to use usePostHog):
       ```typescript
       import { usePathname } from "next/navigation";
       import { useRef } from "react";

       function PostHogPageView() {
         const pathname = usePathname();
         const posthog = usePostHog();
         const prevPathnameRef = useRef<string | null>(null);
         useEffect(() => {
           if (pathname && prevPathnameRef.current !== pathname) {
             prevPathnameRef.current = pathname;
             posthog.capture("$pageview", { $current_url: window.location.href });
           }
         }, [pathname, posthog]);
         return null;
       }
       ```

    c) Wrap the AppProviders return with PostHogProvider:
       ```typescript
       return (
         <PostHogProvider client={posthog}>
           <PostHogPageView />
           <AuthProvider baseUrl={baseUrl} tokenStorage={tokenStorage}>
             ...existing tree...
           </AuthProvider>
         </PostHogProvider>
       );
       ```
       PostHogProvider is the outermost wrapper so PostHogPageView can use usePostHog().

         useEffect(() => {
