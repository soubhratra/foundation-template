**Step 1: Create `apps/accounts/permissions.py`.**

```python
"""Custom permission functions for django-hijack impersonation.

Referenced via HIJACK_PERMISSION_CHECK in settings.
"""


def staff_can_hijack_regular_users(*, hijacker, hijacked) -> bool:
    """Staff can impersonate only non-staff, non-superuser, active accounts.

    Called by django-hijack before every hijack attempt. Returns True if
    hijacker is allowed to impersonate hijacked.

    Rules:
    - hijacked must be active
    - hijacker must be staff or superuser (extra safety)
    - hijacked must NOT be staff or superuser
    """
    if not hijacked.is_active:
        return False
    if not (hijacker.is_staff or hijacker.is_superuser):
        return False
    if hijacked.is_staff or hijacked.is_superuser:
        return False
    return True
```

```python
# B2B_START — ImpersonationEvent (plain model, not pghistory-tracked)
class ImpersonationEvent(models.Model):
    """Records every hijack start and end event.

    Written by hijack_started / hijack_ended signal handlers in apps.audit.apps.
    Plain Django model — NOT pghistory-tracked (hijack events are session ops,
    not ORM model mutations, so pghistory triggers would never fire for them).
    """

    class Action(models.TextChoices):
        START = "start", "Start"
        END = "end", "End"

    action = models.CharField(max_length=10, choices=Action.choices)
    hijacker = models.ForeignKey(
        "accounts.User",
        on_delete=models.SET_NULL,
        null=True,
        related_name="hijack_sessions_as_hijacker",
    )
    hijacked = models.ForeignKey(
        "accounts.User",
        on_delete=models.SET_NULL,
        null=True,
        related_name="hijack_sessions_as_hijacked",
    )
    timestamp = models.DateTimeField(auto_now_add=True)

    class Meta:
        db_table = "audit_impersonation_event"
        ordering = ["-timestamp"]
        verbose_name = "Impersonation Event"
        verbose_name_plural = "Impersonation Events"

    def __str__(self) -> str:
        return f"{self.action} — {self.hijacker} → {self.hijacked} at {self.timestamp}"
# B2B_END
```

"""Audit app configuration."""

from django.apps import AppConfig


class AuditConfig(AppConfig):
    default_auto_field = "django.db.models.BigAutoField"
    name = "apps.audit"
    label = "audit"
    verbose_name = "Audit"

    def ready(self) -> None:
        # B2B_START — hijack signal handlers for impersonation event recording
        try:
            from hijack import signals as hijack_signals
        except ImportError:
            # hijack not installed (non-B2B scaffold) — skip signal wiring
            return

        from apps.audit.models import ImpersonationEvent

        def _on_hijack_started(sender, hijacker, hijacked, request, **kwargs):
            ImpersonationEvent.objects.create(
                action=ImpersonationEvent.Action.START,
                hijacker=hijacker,
                hijacked=hijacked,
            )

        def _on_hijack_ended(sender, hijacker, hijacked, request, **kwargs):
            ImpersonationEvent.objects.create(
                action=ImpersonationEvent.Action.END,
                hijacker=hijacker,
                hijacked=hijacked,
            )

        hijack_signals.hijack_started.connect(_on_hijack_started)
        hijack_signals.hijack_ended.connect(_on_hijack_ended)
        # B2B_END
