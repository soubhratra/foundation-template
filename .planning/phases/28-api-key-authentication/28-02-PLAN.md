def generate_secret() -> tuple[str, str, str]:
    """Return (raw_secret, signing_key, secret_hash)."""
    import hashlib
    import secrets
    token = secrets.token_hex(32)  # 64 hex chars, 256 bits
    raw_secret = token
    signing_key = token[:8]
    secret_hash = hashlib.sha256(raw_secret.encode()).hexdigest()
    return raw_secret, signing_key, secret_hash

**apps/api_keys/models.py** — APIKey model with generate_key() static method. SECURITY RULES (locked in STATE.md):

- Only prefix and key_hash stored — raw key never persisted
- hmac.compare_digest used for hash comparison (NOT ==)
- key_hash is SHA-256 hex of the full raw key (not HMAC)
- unique=True on prefix (display identifier)
- db_index=True on key_hash (auth lookup path)

```python
import hashlib
import secrets
import uuid

from django.conf import settings
from django.db import models


class APIKey(models.Model):
    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    user = models.ForeignKey(
        "accounts.User",
        on_delete=models.CASCADE,
        related_name="api_keys",
    )
    prefix = models.CharField(max_length=32, unique=True)
    key_hash = models.CharField(max_length=64)  # SHA-256 hex — 64 chars
    scopes = models.JSONField(default=list)
    name = models.CharField(max_length=255, blank=True, default="")
    is_active = models.BooleanField(default=True)
    last_used_at = models.DateTimeField(null=True, blank=True)
    usage_count = models.PositiveBigIntegerField(default=0)
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        db_table = "api_keys_key"
        indexes = [
            models.Index(fields=["key_hash"]),           # Auth lookup — high cardinality
            models.Index(fields=["user", "is_active"]),  # List keys for user
            models.Index(fields=["prefix"]),             # Display lookups
        ]

    def __str__(self) -> str:
        return f"{self.prefix} ({self.name or 'unnamed'})"

    @staticmethod
    def generate_key() -> tuple[str, str, str]:
        """Return (raw_key, stored_prefix, key_hash).

        raw_key:        full key shown once to user — NEVER stored
        stored_prefix:  API_KEY_PREFIX + first 8 chars of random (e.g. ft_a1b2c3d4)
        key_hash:       SHA-256 hex of raw_key — stored for auth backend lookup

        Security: 64 hex chars from secrets.token_hex(32) = 256 bits entropy.
        Using secrets module (CSPRNG), not random. Identical approach to Phase 27
        WebhookEndpoint.generate_secret().
        """
        prefix_setting: str = getattr(settings, "API_KEY_PREFIX", "ft_")
        random_part: str = secrets.token_hex(32)  # 64 lowercase hex chars
        raw_key = f"{prefix_setting}{random_part}"
        stored_prefix = f"{prefix_setting}{random_part[:8]}"
        key_hash = hashlib.sha256(raw_key.encode()).hexdigest()
        return raw_key, stored_prefix, key_hash
```

**apps/api_keys/checks.py** — manage.py check validator, mirrors `apps/webhooks/checks.py`:

```python
from django.conf import settings
from django.core.checks import Warning, register


@register()
def check_api_key_scopes(app_configs, **kwargs) -> list:
    errors = []
    scopes = getattr(settings, "API_KEY_SCOPES", None)

    if scopes is None:
        errors.append(
            Warning(
                "API_KEY_SCOPES setting is not defined. "
                "API key creation will fail at runtime.",
                id="api_keys.W001",
            )
        )
        return errors

    if not isinstance(scopes, dict):
        errors.append(
            Warning(
                "API_KEY_SCOPES must be a dict mapping scope name to config dict.",
                id="api_keys.W002",
            )
        )
        return errors

    for scope_name, config in scopes.items():
        if not isinstance(config, dict) or "description" not in config:
            errors.append(
                Warning(
                    f"API_KEY_SCOPES['{scope_name}'] must be a dict with a 'description' key.",
                    hint=f"Got: {config!r}",
                    id="api_keys.W003",
                )
            )

    return errors
```
