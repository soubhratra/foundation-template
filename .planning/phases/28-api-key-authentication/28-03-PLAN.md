```python
import hashlib
import hmac

from django.core.exceptions import PermissionDenied
from django.db import models
from django.http import HttpRequest
from django.utils import timezone
from ninja.security import HttpBearer
from ninja_extra import ControllerBase
from ninja_extra.permissions import BasePermission


class ApiKeyAuth(HttpBearer):
    """Resolve Bearer token to User; set request.api_key for downstream middleware.

    Ninja calls authenticate(request, token) after extracting the token from
    the Authorization: Bearer <token> header. Returns None to fail auth (Ninja
    tries the next configured backend). Returns the User to succeed.

    Side effects on success:
    - Sets request.api_key = api_key (used by RequireScope and rate limit middleware)
    - Sets request.user = api_key.user (IsAuthenticated passes; all user-based logic works)
    - Updates last_used_at and increments usage_count via partial UPDATE (AKEY-03)
    """

    def authenticate(self, request: HttpRequest, token: str) -> object | None:
        key_hash = hashlib.sha256(token.encode()).hexdigest()
        try:
            api_key = (
                # select_related to prevent N+1 on api_key.user access
                __import__("apps.api_keys.models", fromlist=["APIKey"])
                .APIKey.objects.select_related("user")
                .get(key_hash=key_hash, is_active=True)
            )
        except Exception:
            return None

        # Timing-safe comparison — defense-in-depth even though DB lookup already found by hash
        if not hmac.compare_digest(api_key.key_hash, key_hash):
            return None

        # Update usage tracking (AKEY-03) — partial UPDATE avoids lock on unrelated fields
        from apps.api_keys.models import APIKey

        APIKey.objects.filter(pk=api_key.pk).update(
            last_used_at=timezone.now(),
            usage_count=models.F("usage_count") + 1,
        )
        api_key.refresh_from_db(fields=["last_used_at", "usage_count"])

        # Set on request for RequireScope and ApiKeyRateLimitMiddleware
        request.api_key = api_key  # type: ignore[attr-defined]
        request.user = api_key.user  # type: ignore[attr-defined]
        return api_key.user


class RequireScope(BasePermission):
    """Gate access to API key requests that carry the required scope.

    CRITICAL: Rejects session-authenticated requests entirely.
    A request with a valid session cookie but no API key Bearer token
    is rejected with HTTP 403 — there is NO auth fallback.
    This matches success criteria AKEY-01 exactly.

    Usage:
        permissions=[IsAuthenticated, RequireScope("read")]

    IsAuthenticated MUST appear first in the list. RequireScope raises
    PermissionDenied (→ 403) if:
    - request.api_key is absent (session-only request)
    - request.api_key.scopes does not include the required scope
    """

    def __init__(self, scope: str) -> None:
        self.scope = scope

    def has_permission(self, request: HttpRequest, controller: ControllerBase) -> bool:
        api_key = getattr(request, "api_key", None)
        if api_key is None:
            # Session-authenticated request — no API key Bearer token present
            raise PermissionDenied
        if self.scope not in (api_key.scopes or []):
            raise PermissionDenied
        return True
```

Note on the try/except in authenticate(): Using a broad `except Exception` to swallow DoesNotExist and any import error at module load time gracefully — auth backends must never raise, only return None. In production this is `APIKey.DoesNotExist` only. The lazy import pattern avoids circular imports at module load.

Simpler approach using direct import (no circular import at this stage):

```python

                key_hash=key_hash, is_active=True
            )
        except APIKey.DoesNotExist:
```

```python
from datetime import datetime

from django.conf import settings
from ninja import Schema
from pydantic import field_validator


class APIKeyIn(Schema):
    """Request body for POST /me/api-keys/."""
    name: str = ""
    scopes: list[str]

    @field_validator("scopes")
    @classmethod
    def scopes_must_be_valid(cls, v: list[str]) -> list[str]:
        if not v:
            raise ValueError("scopes must contain at least one scope.")
        valid = set(getattr(settings, "API_KEY_SCOPES", {}).keys())
        invalid = set(v) - valid
        if invalid:
            raise ValueError(
                f"Unknown scopes: {sorted(invalid)}. Valid scopes: {sorted(valid)}"
            )
        return v


class APIKeyCreateOut(Schema):
    """Returned ONLY from POST /me/api-keys/ — includes raw key shown once."""
    id: str
    name: str
    scopes: list[str]
    prefix: str       # e.g. "ft_a1b2c3d4" — human-visible identifier in list views
    is_active: bool
    created_at: datetime
    key: str          # Raw key — COPY AND STORE IMMEDIATELY; never shown again


class APIKeyOut(Schema):
    """Returned for list and management operations — NO key field."""
    id: str
    name: str
    scopes: list[str]
    prefix: str
    is_active: bool
    last_used_at: datetime | None

    created_at: datetime
```

**services.py** — thin service layer, no business logic in controllers:

```python
from django.http import Http404

from apps.api_keys.models import APIKey


class APIKeyService:
    @classmethod
    def create(cls, user, name: str, scopes: list[str]) -> tuple[APIKey, str]:
        """Create a new APIKey. Returns (api_key, raw_key).

        raw_key is returned here and NOWHERE ELSE — caller must include in response.
        """
        raw_key, stored_prefix, key_hash = APIKey.generate_key()
        api_key = APIKey.objects.create(
            user=user,
            prefix=stored_prefix,
            key_hash=key_hash,
            scopes=scopes,
            name=name,
            is_active=True,
        )
        return api_key, raw_key

    @classmethod
    def revoke(cls, user, key_id: str) -> None:
        """Revoke an API key by setting is_active=False.

        Raises Http404 if the key does not exist or does not belong to the user.
        """
        try:
            api_key = APIKey.objects.get(id=key_id, user=user)
        except APIKey.DoesNotExist:
            raise Http404
        api_key.is_active = False
        api_key.save(update_fields=["is_active"])

    @classmethod
    def list_for_user(cls, user, page: int = 1, page_size: int = 20) -> dict:
        """Return paginated list of active and inactive keys for user."""
        qs = APIKey.objects.filter(user=user).order_by("-created_at")
        total = qs.count()
        start = (page - 1) * page_size
        results = list(qs[start : start + page_size])
        return {
            "count": total,
            "page": page,
            "page_size": page_size,
            "results": results,
        }
```

```python
from typing import Any

from django.http import HttpRequest
from ninja_extra import ControllerBase, api_controller, route
from ninja_extra.permissions import IsAuthenticated

from apps.api_keys.schemas import APIKeyCreateOut, APIKeyIn, APIKeyOut
from apps.api_keys.services import APIKeyService


@api_controller(
    "/me/api-keys",
    tags=["api-keys"],
    permissions=[IsAuthenticated],
)
class APIKeyController(ControllerBase):
    """Manage API keys for the authenticated user.

    Management endpoints use session auth (web UI). The ApiKeyAuth backend
    is registered separately on product endpoints that accept key-based access.
    """

    @route.post("", response={201: APIKeyCreateOut}, url_name="api_key_create")
    def create_key(self, request: HttpRequest, body: APIKeyIn) -> Any:
        """Create a new API key. Raw key returned ONCE — copy and store immediately."""
        api_key, raw_key = APIKeyService.create(
            user=request.user,
            name=body.name,
            scopes=body.scopes,
        )
        return 201, {
            "id": str(api_key.id),
            "name": api_key.name,
            "scopes": api_key.scopes,
            "prefix": api_key.prefix,
            "is_active": api_key.is_active,
            "created_at": api_key.created_at,
            "key": raw_key,
        }

    @route.get("", response={200: dict}, url_name="api_key_list")
    def list_keys(self, request: HttpRequest, page: int = 1, page_size: int = 20) -> Any:
        """List all API keys for the authenticated user. Raw key is never returned here."""
        result = APIKeyService.list_for_user(
            user=request.user, page=page, page_size=page_size
        )
        # Serialize results using APIKeyOut (no key field)
        result["results"] = [
            APIKeyOut(
                id=str(k.id),
                name=k.name,
                scopes=k.scopes,
                prefix=k.prefix,
                is_active=k.is_active,
                last_used_at=k.last_used_at,
                usage_count=k.usage_count,
                created_at=k.created_at,
            ).model_dump()
            for k in result["results"]
        ]
        return 200, result

    @route.delete("/{key_id}", response={204: None}, url_name="api_key_revoke")
    def revoke_key(self, request: HttpRequest, key_id: str) -> Any:
        """Revoke an API key. Revoked keys cannot authenticate requests."""
        APIKeyService.revoke(user=request.user, key_id=key_id)
        return 204, None
```
