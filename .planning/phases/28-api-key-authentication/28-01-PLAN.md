```python
"""Tests for APIKey model and generate_key() static method.

Requirements: AKEY-01 (key creation — raw key shown once, prefix + hash stored).
"""
import hashlib

import pytest

from apps.api_keys.models import APIKey


@pytest.mark.django_db
class TestAPIKeyGeneration:
    def test_generate_key_returns_three_tuple(self):
        raw_key, prefix, key_hash = APIKey.generate_key()
        assert isinstance(raw_key, str)
        assert isinstance(prefix, str)
        assert isinstance(key_hash, str)

    def test_raw_key_starts_with_prefix_setting(self, settings):
        settings.API_KEY_PREFIX = "ft_"  # pragma: allowlist secret
        raw_key, prefix, key_hash = APIKey.generate_key()
        assert raw_key.startswith("ft_")

    def test_stored_prefix_is_prefix_plus_8_chars(self, settings):
        settings.API_KEY_PREFIX = "ft_"  # pragma: allowlist secret
        raw_key, stored_prefix, key_hash = APIKey.generate_key()
        # stored_prefix = "ft_" + first 8 chars of random = 11 chars total
        assert stored_prefix.startswith("ft_")
        assert len(stored_prefix) == len("ft_") + 8

    def test_key_hash_is_sha256_of_raw_key(self):
        raw_key, prefix, key_hash = APIKey.generate_key()
        expected_hash = hashlib.sha256(raw_key.encode()).hexdigest()
        assert key_hash == expected_hash

    def test_raw_key_not_stored_in_db(self, api_user):
        raw_key, stored_prefix, key_hash = APIKey.generate_key()
        key = APIKey.objects.create(
            user=api_user,
            prefix=stored_prefix,
            key_hash=key_hash,
            scopes=["read"],
            name="Test",
        )
        key.refresh_from_db()
        # raw_key must NOT be a field on the model
        assert not hasattr(key, "raw_key")
        # key_hash stored as SHA-256 hex (64 chars)
        assert len(key.key_hash) == 64

    def test_two_generated_keys_have_different_prefixes(self):
        _, prefix1, _ = APIKey.generate_key()
        _, prefix2, _ = APIKey.generate_key()
        assert prefix1 != prefix2
```

```python
"""Tests for ApiKeyRateLimitMiddleware — per-key rate limiting and X-RateLimit-* headers.

Requirements: AKEY-02.

CRITICAL: test_isolation_settings autouse fixture globally sets RATELIMIT_ENABLE=False.
All tests in this file must override it with settings.RATELIMIT_ENABLE = True.
Use LocMemCache as the cache backend (already configured by test_isolation_settings).
"""
import pytest
from django.test import RequestFactory, override_settings
from django.http import HttpResponse


@pytest.mark.django_db
class TestApiKeyRateLimitHeaders:
    @override_settings(RATELIMIT_ENABLE=True)
    def test_headers_on_normal_response(self, api_key):
        """Normal (non-rate-limited) responses include all three X-RateLimit-* headers."""
        from apps.api_keys.middleware import ApiKeyRateLimitMiddleware

        factory = RequestFactory()
        request = factory.get("/api/v1/some-endpoint/")
        request.api_key = api_key  # type: ignore[attr-defined]

        def get_response(req):
            return HttpResponse("ok", status=200)

        middleware = ApiKeyRateLimitMiddleware(get_response)
        response = middleware(request)

        assert "X-RateLimit-Limit" in response
        assert "X-RateLimit-Remaining" in response
        assert "X-RateLimit-Reset" in response
        assert response.status_code == 200

    @override_settings(RATELIMIT_ENABLE=True)
    def test_rate_limit_returns_429_with_headers(self, api_key, settings):
        """Requests exceeding the rate limit return 429 with X-RateLimit-* headers."""
        from django.core.cache import cache
        from apps.api_keys.middleware import ApiKeyRateLimitMiddleware

        # Set a very low rate limit for testing
        settings.API_KEY_RATE_LIMIT = "1/m"  # pragma: allowlist secret
        cache.clear()

        factory = RequestFactory()

        def get_response(req):
            return HttpResponse("ok", status=200)

        middleware = ApiKeyRateLimitMiddleware(get_response)

        # First request should pass
        request1 = factory.get("/api/v1/some-endpoint/")
        request1.api_key = api_key  # type: ignore[attr-defined]
        response1 = middleware(request1)
        assert response1.status_code == 200

        # Second request should be rate-limited
        request2 = factory.get("/api/v1/some-endpoint/")
        request2.api_key = api_key  # type: ignore[attr-defined]
        response2 = middleware(request2)
        assert response2.status_code == 429
        assert "X-RateLimit-Limit" in response2
        assert "X-RateLimit-Remaining" in response2
        assert "X-RateLimit-Reset" in response2
        assert response2["X-RateLimit-Remaining"] == "0"

    @override_settings(RATELIMIT_ENABLE=True)
    def test_no_headers_on_session_request(self):
        """Session-authenticated requests (no api_key) pass through without RL headers."""
        from apps.api_keys.middleware import ApiKeyRateLimitMiddleware

        factory = RequestFactory()
        request = factory.get("/api/v1/some-endpoint/")
        # No request.api_key set

        def get_response(req):
            return HttpResponse("ok", status=200)

        middleware = ApiKeyRateLimitMiddleware(get_response)
        response = middleware(request)
        assert response.status_code == 200
        assert "X-RateLimit-Limit" not in response
```

```python
"""Tests for APIKeyController HTTP endpoints: create, list, revoke.

Requirements: AKEY-01 (create/revoke lifecycle, raw key shown once).
"""
import pytest
from django.test import Client
from django.contrib.auth import get_user_model

User = get_user_model()


@pytest.mark.django_db
class TestAPIKeyController:
    def _login(self, client: Client, user) -> None:
        client.force_login(user)

    def test_create_key_returns_raw_key_once(self, client, api_user):
        """POST /api/v1/me/api-keys/ returns raw key in response body."""
        self._login(client, api_user)
        response = client.post(
            "/api/v1/me/api-keys/",
            data={"name": "My Key", "scopes": ["read"]},
            content_type="application/json",
        )
        assert response.status_code == 201
        data = response.json()
        # Raw key must be present in create response
        assert "key" in data
        assert data["key"].startswith("ft_")
        # Prefix stored separately
        assert "prefix" in data
        assert data["prefix"] != data["key"]

    def test_list_keys_does_not_include_raw_key(self, client, api_user, api_key):
        """GET /api/v1/me/api-keys/ does not expose raw key field."""
        self._login(client, api_user)
        response = client.get("/api/v1/me/api-keys/")
        assert response.status_code == 200
        data = response.json()
        results = data.get("results", data) if isinstance(data, dict) else data
        for key_record in results:
            assert "key" not in key_record

    def test_revoke_key_sets_inactive(self, client, api_user, api_key):
        """DELETE /api/v1/me/api-keys/{id}/ sets is_active=False."""
        from apps.api_keys.models import APIKey

        self._login(client, api_user)
        response = client.delete(f"/api/v1/me/api-keys/{api_key.id}/")
        assert response.status_code == 204
        api_key.refresh_from_db()
        assert api_key.is_active is False

    def test_revoked_key_auth_returns_401(self, client, api_user, revoked_api_key):
        """Attempting to use a revoked key for authentication returns 401."""
        response = client.get(
            "/api/v1/me/api-keys/",
            HTTP_AUTHORIZATION=f"Bearer {revoked_api_key._raw_key}",
        )
        assert response.status_code in (401, 403)

    def test_create_key_invalid_scope_returns_422(self, client, api_user):
        """POST with an unknown scope returns 422 validation error."""
        self._login(client, api_user)
        response = client.post(
            "/api/v1/me/api-keys/",
            data={"name": "Bad Key", "scopes": ["nonexistent_scope"]},
            content_type="application/json",
        )
        assert response.status_code == 422

    def test_create_key_empty_scopes_returns_422(self, client, api_user):
        """POST with empty scopes list returns 422."""
        self._login(client, api_user)
        response = client.post(
            "/api/v1/me/api-keys/",
            data={"name": "Empty", "scopes": []},
            content_type="application/json",
        )
        assert response.status_code == 422

    def test_list_keys_shows_usage_count_and_last_used(self, client, api_user, api_key):
        """GET response includes usage_count and last_used_at fields."""
        self._login(client, api_user)
        response = client.get("/api/v1/me/api-keys/")
        assert response.status_code == 200
        data = response.json()
        results = data.get("results", data) if isinstance(data, dict) else data
        if results:
            record = results[0]
            assert "usage_count" in record
            assert "last_used_at" in record
```
