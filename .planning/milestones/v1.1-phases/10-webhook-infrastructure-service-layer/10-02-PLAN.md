    ```python
    """
    BillingService — all billing operations. Controllers call this; no ORM or adapter in controllers.

    Usage:
        service = BillingService()          # uses BILLING_ADAPTER from settings
        service = BillingService(adapter=stub)  # testable without Stripe
    """
    import logging
    from django.core.exceptions import PermissionDenied
    from apps.billing.models import Customer, Subscription, Payment, PlanProviderMapping, SubscriptionStatus
    from apps.core.exceptions import ServiceError

    logger = logging.getLogger(__name__)

    class BillingService:
        def __init__(self, adapter=None):
            if adapter is None:
                from .utils import get_adapter
                self._adapter = get_adapter()
            else:
                self._adapter = adapter

        def _get_customer(self, user) -> Customer:
            try:
                return user.billing_customer
            except Customer.DoesNotExist:
                raise ServiceError(code="no_customer", message="No billing customer record found.", status_code=404)

        def _resolve_price_id(self, plan_id: str, billing_period: str) -> str:
            try:
                mapping = PlanProviderMapping.objects.get(
                    plan_id=plan_id,
                    provider=self._get_provider_name(),
                    billing_period=billing_period,
                    is_active=True,
                )
                return mapping.price_id
            except PlanProviderMapping.DoesNotExist:
                raise ServiceError(code="no_price_mapping", message=f"No price mapping for plan={plan_id} period={billing_period}", status_code=400)

        def _get_provider_name(self) -> str:
            # Derive provider name from adapter class name: StripeAdapter -> "stripe"
            return type(self._adapter).__name__.lower().replace("adapter", "")

        def create_checkout_session(self, user, plan_id: str, billing_period: str, success_url: str, cancel_url: str, trial_days: int = 0) -> dict:
            customer = self._get_customer(user)
            price_id = self._resolve_price_id(plan_id, billing_period)
            return self._adapter.create_checkout_session(
                customer_id=customer.stripe_customer_id,
                price_id=price_id,
                mode="subscription",
                success_url=success_url,
                cancel_url=cancel_url,
                trial_days=trial_days,
            )

        def upgrade_subscription(self, user, new_plan_id: str, billing_period: str = "monthly") -> dict:
            customer = self._get_customer(user)
            sub = customer.subscription
            new_price_id = self._resolve_price_id(new_plan_id, billing_period)
            return self._adapter.modify_subscription(
                subscription_id=sub.stripe_subscription_id,
                new_price_id=new_price_id,
                proration_behavior="create_prorations",
            )

        def downgrade_subscription(self, user, new_plan_id: str, billing_period: str = "monthly") -> dict:
            customer = self._get_customer(user)
            sub = customer.subscription
            new_price_id = self._resolve_price_id(new_plan_id, billing_period)
            return self._adapter.modify_subscription(
                subscription_id=sub.stripe_subscription_id,
                new_price_id=new_price_id,
                proration_behavior="always_invoice",
            )

        def cancel_subscription(self, user, cancel_at_period_end: bool = True) -> dict:
            customer = self._get_customer(user)
            sub = customer.subscription
            result = self._adapter.cancel_subscription(
                subscription_id=sub.stripe_subscription_id,
                cancel_at_period_end=cancel_at_period_end,
            )
            sub.cancel_at_period_end = cancel_at_period_end
            sub.save(update_fields=["cancel_at_period_end", "updated_at"])
            return result

        def reactivate_subscription(self, user, plan_id: str, billing_period: str = "monthly") -> dict:
            customer = self._get_customer(user)
            sub = customer.subscription
            new_price_id = self._resolve_price_id(plan_id, billing_period)
            return self._adapter.modify_subscription(
                subscription_id=sub.stripe_subscription_id,
                new_price_id=new_price_id,
                proration_behavior="create_prorations",
            )

        def create_one_time_payment(self, user, success_url: str, cancel_url: str, price_id: str) -> dict:
            customer = self._get_customer(user)
            return self._adapter.create_checkout_session(
                customer_id=customer.stripe_customer_id,
                price_id=price_id,
                mode="payment",
                success_url=success_url,
                cancel_url=cancel_url,
            )

        def record_usage(self, user, meter_event_name: str, value: int, identifier: str | None = None) -> dict:
            customer = self._get_customer(user)
            return self._adapter.record_usage(
                meter_event_name=meter_event_name,
                customer_id=customer.stripe_customer_id,
                value=value,
                identifier=identifier,
            )

        def get_usage_summary(self, user, subscription_item_id: str) -> dict:
            return self._adapter.get_usage_summary(subscription_item_id=subscription_item_id)

        def preview_proration(self, user, new_plan_id: str, billing_period: str = "monthly") -> dict:
            customer = self._get_customer(user)
            sub = customer.subscription
            new_price_id = self._resolve_price_id(new_plan_id, billing_period)
            return self._adapter.preview_proration(
                subscription_id=sub.stripe_subscription_id,
                new_price_id=new_price_id,
            )

        def create_portal_session(self, user, return_url: str) -> dict:
            customer = self._get_customer(user)
            return self._adapter.create_portal_session(
                customer_id=customer.stripe_customer_id,
                return_url=return_url,
            )

        def list_invoices(self, user, limit: int = 10) -> list[dict]:
            customer = self._get_customer(user)
            return self._adapter.list_invoices(
                customer_id=customer.stripe_customer_id,
                limit=limit,
            )
    ```
