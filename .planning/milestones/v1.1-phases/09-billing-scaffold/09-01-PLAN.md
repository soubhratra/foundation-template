    """
    Tests for PaymentAdapter ABC and StripeAdapter implementation.
    Covers: ADPTR-01 (ABC contract), ADPTR-02 (StripeAdapter), ADPTR-03 (settings resolution)
    """
    import abc
    import pytest

    adapters = pytest.importorskip("apps.billing.adapters")
    PaymentAdapter = adapters.PaymentAdapter
    StripeAdapter = adapters.StripeAdapter


    class TestPaymentAdapterContract:
        """ADPTR-01: PaymentAdapter is an ABC with domain-centric method signatures."""

        def test_payment_adapter_is_abstract(self):
            assert abc.ABC in PaymentAdapter.__mro__, "PaymentAdapter must be an ABC"

        def test_cannot_instantiate_abstract_adapter(self):
            with pytest.raises(TypeError):
                PaymentAdapter()  # type: ignore[abstract]

        def test_abstract_methods_defined(self):
            expected = {
                "create_customer",
                "create_checkout_session",
                "create_portal_session",
                "cancel_subscription",
                "modify_subscription",
                "construct_webhook_event",
                "record_usage",
                "list_invoices",
            }
            abstract_methods = getattr(PaymentAdapter, "__abstractmethods__", set())
            assert expected == abstract_methods, (
                f"Abstract methods mismatch.\nExpected: {expected}\nGot: {abstract_methods}"
            )

        def test_no_stripe_in_abstract_signatures(self):
            """ADPTR-01: The word 'stripe' must not appear in abstract method signatures."""
            import inspect

            for name in PaymentAdapter.__abstractmethods__:
                method = getattr(PaymentAdapter, name)
                sig = str(inspect.signature(method))
                annotations = getattr(method, "__annotations__", {})
                sig_lower = sig.lower()
                assert "stripe" not in sig_lower, (
                    f"Abstract method '{name}' contains 'stripe' in signature: {sig}"
                )
                for annotation_value in annotations.values():
                    annotation_str = str(annotation_value).lower()
                    assert "stripe" not in annotation_str, (
                        f"Abstract method '{name}' has 'stripe' in return/param annotation"
                    )


    class TestStripeAdapterImplementation:
        """ADPTR-02: StripeAdapter concretely implements PaymentAdapter."""

        def test_stripe_adapter_inherits_payment_adapter(self):
            assert issubclass(StripeAdapter, PaymentAdapter)

        def test_stripe_adapter_is_concrete(self):
            """StripeAdapter must have no abstract methods remaining."""
            abstract = getattr(StripeAdapter, "__abstractmethods__", set())
            assert not abstract, (
                f"StripeAdapter has unimplemented abstract methods: {abstract}"
            )

        def test_stripe_adapter_instantiates(self):
            """StripeAdapter can be constructed (uses STRIPE_SECRET_KEY from settings)."""
            from django.conf import settings

            if not getattr(settings, "STRIPE_SECRET_KEY", ""):
                pytest.skip("STRIPE_SECRET_KEY not set — skipping instantiation test")
            adapter = StripeAdapter()
            assert isinstance(adapter, PaymentAdapter)


    class TestAdapterResolution:
        """ADPTR-03: Adapter resolved via BILLING_ADAPTER setting."""

        def test_billing_adapter_setting_exists(self):
            from django.conf import settings

            assert hasattr(settings, "BILLING_ADAPTER"), "BILLING_ADAPTER setting must exist"
            assert settings.BILLING_ADAPTER, "BILLING_ADAPTER must not be empty"

        def test_billing_adapter_resolves_to_stripe_adapter(self):
            from django.utils.module_loading import import_string
            from django.conf import settings

            klass = import_string(settings.BILLING_ADAPTER)
            assert issubclass(klass, PaymentAdapter)
    ```

    """
    Tests for billing data models.
    Covers: CUST-01 (Customer model), SUB-06 (SubscriptionStatus + audit fields), PAY-02 (Payment model)
    """
    import pytest
    from decimal import Decimal

    billing_models = pytest.importorskip("apps.billing.models")
    Customer = billing_models.Customer
    Subscription = billing_models.Subscription
    Payment = billing_models.Payment
    ProcessedWebhookEvent = billing_models.ProcessedWebhookEvent
    SubscriptionStatus = billing_models.SubscriptionStatus


    class TestCustomerModel:
        """CUST-01: Customer model with OneToOne User link."""

        def test_customer_created_with_user(self, customer, user):
            assert customer.user == user

        def test_customer_stripe_customer_id_blank_by_default(self, customer):
            """stripe_customer_id is populated lazily — blank on creation is valid."""
            assert customer.stripe_customer_id == "" or customer.stripe_customer_id is None

        def test_customer_has_timestamps(self, customer):
            assert customer.created_at is not None
            assert customer.updated_at is not None

        def test_customer_str(self, customer, user):
            assert str(user.email) in str(customer)


    class TestSubscriptionStatus:
        """SUB-06: SubscriptionStatus is a Django TextChoices enum with full Stripe-matching values."""

        def test_required_statuses_exist(self):
            required = {
                "active", "past_due", "canceled", "trialing",
                "incomplete", "incomplete_expired", "unpaid",
            }
            actual = {choice.value for choice in SubscriptionStatus}
            assert required == actual, (
                f"SubscriptionStatus values mismatch.\nExpected: {required}\nGot: {actual}"
            )

        def test_is_text_choices(self):
            from django.db.models import TextChoices
            assert issubclass(SubscriptionStatus, TextChoices)


    class TestSubscriptionModel:
        """SUB-06: Subscription model audit fields and cancel_at_period_end."""

        def test_subscription_has_cancel_at_period_end(self):
            field_names = [f.name for f in Subscription._meta.get_fields()]
            assert "cancel_at_period_end" in field_names

        def test_subscription_has_period_timestamps(self):
            field_names = [f.name for f in Subscription._meta.get_fields()]
            assert "current_period_start" in field_names
            assert "current_period_end" in field_names

        def test_subscription_status_uses_enum(self):
            status_field = Subscription._meta.get_field("status")
            assert status_field.choices, "status field must have choices from SubscriptionStatus"


    class TestPaymentModel:
        """PAY-02: Payment model records one-time payment with amount and status."""

        def test_payment_has_amount_field(self):
            from django.db.models import DecimalField
            amount_field = Payment._meta.get_field("amount")
            assert isinstance(amount_field, DecimalField)

        def test_payment_has_required_fields(self):
            field_names = [f.name for f in Payment._meta.get_fields()]
            for required in ["stripe_payment_intent_id", "amount", "currency", "status"]:
                assert required in field_names, f"Payment model missing field: {required}"


    class TestProcessedWebhookEvent:
        """HOOK-03 prerequisite: ProcessedWebhookEvent unique_together constraint."""

        def test_unique_together_provider_event_id(self):
            unique_together = ProcessedWebhookEvent._meta.unique_together
            # Check either unique_together or UniqueConstraint covers (provider, event_id)
            field_set = {"provider", "event_id"}
            covered = any(set(combo) == field_set for combo in unique_together)
            if not covered:
                constraints = [
                    set(c.fields) for c in ProcessedWebhookEvent._meta.constraints
                    if hasattr(c, "fields")
                ]
                covered = field_set in constraints
            assert covered, "ProcessedWebhookEvent must have unique_together on (provider, event_id)"

    """
    Tests for billing signals.
    Covers: CUST-02 (Customer auto-created on User post_save)
    """
    import pytest
    from django.contrib.auth import get_user_model

    pytest.importorskip("apps.billing.signals")
    billing_models = pytest.importorskip("apps.billing.models")
    Customer = billing_models.Customer

    User = get_user_model()


    @pytest.mark.django_db
    def test_customer_auto_created_on_user_creation():
        """CUST-02: Creating a User automatically creates a linked Customer record."""
        user = User.objects.create_user(
            username="signaltest",
            email="signaltest@example.com",
            password="testpass123",
        )
        assert Customer.objects.filter(user=user).exists(), (
            "Customer must be auto-created via post_save signal when User is created"
        )


    @pytest.mark.django_db
    def test_customer_not_duplicated_on_user_save():
        """CUST-02: Saving an existing User does not create a second Customer."""
        user = User.objects.create_user(
            username="signaltest2",
            email="signaltest2@example.com",
            password="testpass123",
        )
        user.first_name = "Updated"
        user.save()
        count = Customer.objects.filter(user=user).count()
        assert count == 1, f"Expected 1 Customer, got {count} after User re-save"
    ```
