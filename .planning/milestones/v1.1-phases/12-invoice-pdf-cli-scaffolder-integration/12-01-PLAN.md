       @route.post("/invoices/{payment_id}/pdf", response=PdfTaskOut, url_name="billing-invoice-pdf")
       def request_invoice_pdf(self, request, payment_id: int):
           """Enqueue PDF generation. Returns task_id for polling. Never blocks."""
           invoice_data = self.service.get_payment_for_pdf(request.user, payment_id)
           task = generate_invoice_pdf.delay(invoice_data)
           return PdfTaskOut(task_id=task.id, status="pending")

       @route.get("/invoices/pdf-status/{task_id}", url_name="billing-pdf-status")
       def get_pdf_status(self, request, task_id: str):
           """Poll PDF generation. When status=success, returns base64-encoded PDF bytes."""
           from celery.result import AsyncResult
           import base64
           result = AsyncResult(task_id)
           if result.state == "SUCCESS":
               pdf_bytes = result.result
               return {"status": "success", "pdf_base64": base64.b64encode(pdf_bytes).decode()}
           elif result.state == "FAILURE":
               return api.create_response(request, {"status": "failed"}, status=500)
           return {"status": result.state.lower()}

       NOTE: The polling endpoint returns different shapes depending on state — omit `response=`
       type annotation on the polling endpoint and return dicts directly (Django Ninja allows
       untyped returns — it will serialize the dict as JSON).
