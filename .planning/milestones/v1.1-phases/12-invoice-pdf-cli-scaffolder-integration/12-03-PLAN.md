    ```python
    import subprocess
    import shutil
    from pathlib import Path
    import pytest

    PROJECT_ROOT = Path(__file__).resolve().parents[3]  # workspace/foundation-template/
    CLI_DIST = PROJECT_ROOT / "packages/create-foundation-app/dist/index.js"


    @pytest.mark.skipif(
        not CLI_DIST.exists(),
        reason="CLI not built — run: cd packages/create-foundation-app && pnpm build"
    )
    class TestBillingRemovalViaCLI:
        def test_billing_removal_via_cli(self, tmp_path):
            """SCAFF-04: Invokes the actual compiled CLI with --without billing and verifies
            that all billing artifacts are absent from the output directory.

            Uses subprocess (not Python-reimplemented regex) so any TypeScript bug in
            removeBillingModule() will surface here. Unit-level regex coverage lives in
            packages/create-foundation-app/tests/remove-billing.test.ts (Plan 12-02).
            """
            # Step 1: Copy billing-relevant source files to tmp_path as if it were a new project
            src_api_settings = PROJECT_ROOT / "apps/api/config/settings"
            src_urls = PROJECT_ROOT / "apps/api/config/urls.py"
            src_billing_app = PROJECT_ROOT / "apps/api/apps/billing"
            src_web_billing = PROJECT_ROOT / "apps/web/app/(dashboard)/billing"
            src_billing_config = PROJECT_ROOT / "apps/web/billing.config.ts"

            dest_settings = tmp_path / "apps/api/config/settings"
            dest_settings.mkdir(parents=True)
            shutil.copy2(src_api_settings / "base.py", dest_settings / "base.py")
            shutil.copy2(src_api_settings / "development.py", dest_settings / "development.py")
            shutil.copy2(src_urls, tmp_path / "apps/api/config/urls.py")

            if src_billing_app.exists():
                shutil.copytree(src_billing_app, tmp_path / "apps/api/apps/billing")
            if src_web_billing.exists():
                shutil.copytree(src_web_billing, tmp_path / "apps/web/app/(dashboard)/billing",
                                dirs_exist_ok=True)
            if src_billing_config.exists():
                dest_web = tmp_path / "apps/web"
                dest_web.mkdir(parents=True, exist_ok=True)
                shutil.copy2(src_billing_config, dest_web / "billing.config.ts")

            # Step 2: Invoke the compiled CLI — use a mode that only runs removeBillingModule
            # on the target dir without the full interactive scaffold flow.
            # If CLI supports --remove-only or --skip-scaffold, use that.
            # Otherwise, invoke via Node directly passing the outputDir as an env var or arg:
            result = subprocess.run(
                ["node", str(CLI_DIST), "--without", "billing",
                 "--output-dir", str(tmp_path), "--skip-validation", "--non-interactive"],
                capture_output=True, text=True, timeout=60,
                cwd=str(PROJECT_ROOT / "packages/create-foundation-app")
            )
            # NOTE: If the CLI does not support --non-interactive or --output-dir flags,
            # adapt to whatever flags index.ts actually accepts. Read the CLI's argument
            # parser (check for yargs, commander, or custom argv parsing) and use the
            # appropriate flags. The goal is to trigger removeBillingModule() on tmp_path.

            # If subprocess invocation is not feasible with available CLI flags,
            # fall back to the direct module approach below (see fallback note).
            assert result.returncode == 0, (
                f"CLI exited with code {result.returncode}\n"
                f"stdout: {result.stdout}\nstderr: {result.stderr}"
            )

            # Step 3: Assert billing artifacts are absent
            base_py = (tmp_path / "apps/api/config/settings/base.py").read_text()
            dev_py = (tmp_path / "apps/api/config/settings/development.py").read_text()
            urls_py = (tmp_path / "apps/api/config/urls.py").read_text()

            assert "apps.billing" not in base_py, "apps.billing still in INSTALLED_APPS"
            assert "BILLING MODULE" not in base_py, "BILLING MODULE block not removed from base.py"
            assert "BILLING_ADAPTER" not in dev_py, "BILLING_ADAPTER line not removed from development.py"
            assert "from apps.billing" not in urls_py, "billing import not removed from urls.py"
            assert "BILLING_START" not in urls_py, "BILLING_START marker not removed from urls.py"
            assert not (tmp_path / "apps/api/apps/billing").exists(), "billing app dir still exists"
            assert not (tmp_path / "apps/web/billing.config.ts").exists(), "billing.config.ts still exists"
    ```
