**Part B: Runtime environment variable validation (MONO-06 requirement)**

Missing required env vars must produce clear error messages on startup, not cryptic runtime failures.

3. **Django (apps/api/config/settings/base.py)** -- Django already uses django-environ which provides defaults for all vars. Update the existing base.py to add explicit validation for PRODUCTION mode. Add the following block at the END of base.py (after all settings are defined):

   ```python
   # --------------------------------------------------------------------------
   # Environment Variable Validation
   # --------------------------------------------------------------------------
   # In production, certain env vars MUST be explicitly set (not using defaults).
   # In development, defaults are used for zero-config startup.
   #
   # This validation runs at import time (Django startup) and produces clear
   # error messages identifying exactly which env vars are missing.
   # --------------------------------------------------------------------------
   if not DEBUG:
       _required_in_production = {
           "SECRET_KEY": SECRET_KEY,
           "DATABASE_URL": None,  # checked separately
           "ALLOWED_HOSTS": ALLOWED_HOSTS,
       }

       _missing = []
       if SECRET_KEY == "django-insecure-change-me-in-production":
           _missing.append("SECRET_KEY (still using insecure default)")
       if ALLOWED_HOSTS == ["localhost", "127.0.0.1"]:
           _missing.append("ALLOWED_HOSTS (still using localhost defaults)")

       if _missing:
           import sys
           print(
               "\n"
               "=" * 60 + "\n"
               "MISSING REQUIRED ENVIRONMENT VARIABLES\n"
               "=" * 60 + "\n"
               + "\n".join(f"  - {var}" for var in _missing)
               + "\n\n"
               "Set these in your .env file or environment.\n"
               "See apps/api/.env.example for reference.\n"
               "=" * 60 + "\n",
               file=sys.stderr,
           )
           raise ImproperlyConfigured(
               f"Missing required environment variables: {', '.join(_missing)}"
           )
   ```

   Also add `from django.core.exceptions import ImproperlyConfigured` to the imports if not already there.

4. **Next.js (apps/web/app/env.ts)** -- Runtime check imported by root layout:

   ```typescript
   /**
    * Environment variable validation for Next.js web app.
    *
    * Import this module in app/layout.tsx to validate env vars at startup.
    * Missing required vars will throw a clear error message instead of
    * producing cryptic runtime failures.
    */

   function validateEnv() {
     const required: Record<string, string | undefined> = {
       NEXT_PUBLIC_API_URL: process.env.NEXT_PUBLIC_API_URL,
     };

     const missing = Object.entries(required)
       .filter(([, value]) => !value)
       .map(([key]) => key);

     if (missing.length > 0) {
       throw new Error(
         [
           "",
           "=".repeat(60),
           "MISSING REQUIRED ENVIRONMENT VARIABLES",
           "=".repeat(60),
           ...missing.map((key) => `  - ${key}`),
           "",
           "Copy apps/web/.env.example to apps/web/.env.local",
           "and fill in the required values.",
           "=".repeat(60),
         ].join("\n"),
       );
     }

     return required as Record<string, string>;
   }

   export const env = validateEnv();
   ```

   Then update **apps/web/app/layout.tsx** to import the validation. Add this import near the top of the file:

   ```typescript
   import "@/app/env";
   ```

   This ensures env vars are validated when the app starts. The import is side-effect only -- it throws on missing vars.

   ```json
   {
     "scripts": {
       "dev": "turbo dev",
       "build": "turbo build",
       "lint": "turbo lint",
       "test": "turbo test",
       "type-check": "turbo type-check",

     }
   }
   ```

   Key scripts explained:
   - `dev`: Delegates to `turbo dev` which starts all apps in parallel
   - `db:up` / `db:down`: Start/stop Docker infrastructure
   - `db:migrate` / `db:makemigrations`: Django migration shortcuts
   - `db:reset`: Nuclear option -- destroys volumes and recreates from scratch
   - `setup`: One-command setup for new developers (install deps, start infra, run migrations)
